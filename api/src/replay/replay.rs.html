1//! Dead-Letter Queue Replay Binary
2//!
3//! Reads failed `PromptRequest` records from a dead-letter queue dump (NDJSON
4//! format) and replays them through a running orchestrator's HTTP API.
5//!
6//! ## Usage
7//!
8//! ```text
9//! # Replay a DLQ file (max 3 retries per request):
10//! replay --queue-file dlq.ndjson --orchestrator-url http://localhost:8080
11//!
12//! # Read from stdin, increase retry budget:
13//! cat dlq.ndjson | replay --max-retries 5
14//!
15//! # Dry-run: parse and validate without submitting:
16//! replay --queue-file dlq.ndjson --dry-run
17//! ```
18//!
19//! ## NDJSON Format
20//!
21//! Each line is a JSON object. Fields match the HTTP `POST /api/v1/infer` body:
22//!
23//! ```json
24//! {"prompt":"Hello pipeline","session_id":"s1","metadata":{},"deadline_secs":null}
25//! ```
26//!
27//! The replay tool also accepts the raw `DroppedRequest` format exported from
28//! the `/api/v1/debug/dlq` endpoint:
29//!
30//! ```json
31//! {"request_id":"req-1","session_id":"s1","reason":"backpressure","dropped_at":1711900000}
32//! ```
33//!
34//! When a raw `DroppedRequest` is detected (no `prompt` field), the tool
35//! reconstructs a minimal `PromptRequest` with `input` set to `"<replay>"` and
36//! the original `request_id` preserved in metadata.
37

52/// Replay dead-letter queue entries into a running tokio-prompt-orchestrator.
53#[derive(Debug, Parser)]
54#[command(name = "replay", version, about, long_about = None)]
55struct Args {
56    /// Path to the NDJSON queue file.
57    ///
58    /// When omitted (or set to `-`), entries are read from stdin.
59    #[arg(long, short = 'f', value_name = "FILE")]
60    queue_file: Option<PathBuf>,
61
62    /// Maximum number of retry attempts for each failed request.
63    ///
64    /// Requests that fail more than this many times are skipped and counted
65    /// in the final error summary.  Default: 3.
66    #[arg(long, default_value_t = 3, value_name = "N")]
67    max_retries: u32,
68
69    /// Base URL of the running orchestrator, e.g. `http://localhost:8080`.
70    #[arg(
71        long,
72        default_value = "http://127.0.0.1:8080",
73        value_name = "URL",
74        env = "ORCHESTRATOR_URL"
75    )]
76    orchestrator_url: String,
77
78    /// Optional API key sent as `Authorization: Bearer <key>`.
79    #[arg(long, value_name = "KEY", env = "API_KEY")]
80    api_key: Option<String>,
81
82    /// Parse and validate the NDJSON without submitting any requests.
83    #[arg(long, default_value_t = false)]
84    dry_run: bool,
85
86    /// Delay between submissions (milliseconds) to avoid overwhelming the pipeline.
87    #[arg(long, default_value_t = 50, value_name = "MS")]
88    delay_ms: u64,
89
90    /// Timeout for each individual HTTP request (seconds).
91    #[arg(long, default_value_t = 30, value_name = "SECS")]
92    request_timeout_secs: u64,
93}

99/// A single infer request — matches the `POST /api/v1/infer` body.
100#[derive(Debug, Clone, Serialize, Deserialize)]
101struct ReplayEntry {
102    /// The prompt to submit.
103    #[serde(default)]
104    prompt: String,
105    /// Optional session identifier.
106    #[serde(default)]
107    session_id: Option<String>,
108    /// Arbitrary metadata forwarded through the pipeline.
109    #[serde(default)]
110    metadata: HashMap<String, String>,
111    /// Optional deadline in seconds from now.
112    #[serde(default)]
113    deadline_secs: Option<u64>,
114}
115
116/// Raw `DroppedRequest` shape from the `/api/v1/debug/dlq` endpoint.
117///
118/// Detected when the `prompt` field is absent.
119#[derive(Debug, Deserialize)]
120struct DroppedRequestEntry {
121    request_id: String,
122    session_id: String,
123    reason: String,
124}
125
126/// Response body from `POST /api/v1/infer`.
127#[derive(Debug, Deserialize)]
128struct InferResponse {
129    request_id: String,
130    status: String,
131    #[serde(default)]
132    #[allow(dead_code)] // part of the response schema; not printed by replay
133    error: Option<String>,
134}

136// ============================================================================
137// Progress display
138// ============================================================================
139
140/// Simple terminal progress bar that prints to stderr.
141///
142/// Uses only the standard library — no external TUI dependency needed for this
143/// lightweight binary.  For richer display, the `tui` feature binary provides
144/// a full Ratatui dashboard.
145struct ProgressBar {
146    total: usize,
147    current: usize,
148    succeeded: usize,
149    failed: usize,
150    skipped: usize,
151}
152
153impl ProgressBar {
154    fn new(total: usize) -> Self {
155        Self {
156            total,
157            current: 0,
158            succeeded: 0,
159            failed: 0,
160            skipped: 0,
161        }
162    }
163
164    /// Advance by one item and print the updated bar.
165    fn advance(&mut self, succeeded: bool, skipped: bool) {
166        self.current += 1;
167        if skipped {
168            self.skipped += 1;
169        } else if succeeded {
170            self.succeeded += 1;
171        } else {
172            self.failed += 1;
173        }
174        self.render();
175    }
176
177    fn render(&self) {
178        let bar_width: usize = 40;
179        let filled = (self.current * bar_width)
180            .checked_div(self.total)
181            .unwrap_or(0);
182        let empty = bar_width.saturating_sub(filled);
183        let bar: String = format!(
184            "[{}{}] {}/{} ok:{} fail:{} skip:{}",
185            "#".repeat(filled),
186            "-".repeat(empty),
187            self.current,
188            self.total,
189            self.succeeded,
190            self.failed,
191            self.skipped,
192        );
193        // Use \r to overwrite the line on terminals; newline on the last item.
194        if self.current >= self.total {
195            eprintln!("\r{bar}");
196        } else {
197            eprint!("\r{bar}");
198        }
199    }
200
201    fn print_summary(&self) {
202        eprintln!(
203            "\nReplay complete: {} submitted, {} succeeded, {} failed, {} skipped.",
204            self.current, self.succeeded, self.failed, self.skipped
205        );
206    }
207}

213/// Attempt to submit `entry` to the orchestrator with up to `max_retries`
214/// retry attempts using exponential back-off.
215///
216/// Returns `true` on success, `false` if all attempts are exhausted.
217async fn submit_with_retry(
218    client: &Client,
219    endpoint: &str,
220    entry: &ReplayEntry,
221    api_key: Option<&str>,
222    max_retries: u32,
223) -> bool {
224    let mut attempt = 0u32;
225    loop {
226        let mut req = client.post(endpoint).json(entry);
227        if let Some(key) = api_key {
228            req = req.bearer_auth(key);
229        }
230
231        match req.send().await {
232            Ok(resp) if resp.status().is_success() => {
233                // Optionally parse the response to log the request_id.
234                if let Ok(body) = resp.json::<InferResponse>().await {
235                    eprintln!(
236                        "  -> accepted: request_id={} status={}",
237                        body.request_id, body.status
238                    );
239                }
240                return true;
241            }
242            Ok(resp) => {
243                let status = resp.status();
244                let body = resp
245                    .text()
246                    .await
247                    .unwrap_or_else(|_| "<unreadable>".to_string());
248                eprintln!(
249                    "  attempt {}/{}: HTTP {} — {}",
250                    attempt + 1,
251                    max_retries + 1,
252                    status,
253                    body.chars().take(120).collect::<String>()
254                );
255            }
256            Err(e) => {
257                eprintln!(
258                    "  attempt {}/{}: network error — {}",
259                    attempt + 1,
260                    max_retries + 1,
261                    e
262                );
263            }
264        }
265
266        attempt += 1;
267        if attempt > max_retries {
268            return false;
269        }
270        // Exponential back-off: 100ms, 200ms, 400ms, …
271        let delay = Duration::from_millis(100u64.saturating_mul(1u64 << attempt.min(10)));
272        tokio::time::sleep(delay).await;
273    }
274}

276/// Parse a single NDJSON line into a [`ReplayEntry`].
277///
278/// Handles both the full `InferRequest` shape (has `prompt`) and the raw
279/// `DroppedRequest` shape (has `request_id` + `reason`).
280fn parse_line(line: &str) -> Option<ReplayEntry> {
281    let line = line.trim();
282    if line.is_empty() || line.starts_with('#') {
283        return None; // skip blank lines and comments
284    }
285
286    // Try full InferRequest shape first.
287    if let Ok(entry) = serde_json::from_str::<ReplayEntry>(line) {
288        if !entry.prompt.is_empty() {
289            return Some(entry);
290        }
291    }
292
293    // Fallback: parse as DroppedRequest and reconstruct a minimal entry.
294    if let Ok(dropped) = serde_json::from_str::<DroppedRequestEntry>(line) {
295        let mut metadata = HashMap::new();
296        metadata.insert("original_request_id".to_string(), dropped.request_id);
297        metadata.insert("dlq_reason".to_string(), dropped.reason);
298        return Some(ReplayEntry {
299            prompt: "<replay>".to_string(),
300            session_id: Some(dropped.session_id),
301            metadata,
302            deadline_secs: None,
303        });
304    }
305
306    eprintln!("  [warn] could not parse line, skipping: {}", &line[..line.len().min(80)]);
307    None
308}

314#[tokio::main]
315async fn main() -> Result<(), Box<dyn std::error::Error>> {
316    // Initialise a simple env-filter tracing subscriber for the replay binary.
317    tracing_subscriber::fmt()
318        .with_env_filter(
319            tracing_subscriber::EnvFilter::try_from_default_env()
320                .unwrap_or_else(|_| tracing_subscriber::EnvFilter::new("warn")),
321        )
322        .with_target(false)
323        .init();
324
325    let args = Args::parse();
326
327    let infer_endpoint = format!("{}/api/v1/infer", args.orchestrator_url.trim_end_matches('/'));
328
329    eprintln!("tokio-prompt-orchestrator DLQ Replay");
330    eprintln!("  endpoint   : {infer_endpoint}");
331    eprintln!("  max-retries: {}", args.max_retries);
332    eprintln!("  dry-run    : {}", args.dry_run);
333    eprintln!("  delay-ms   : {}", args.delay_ms);
334
335    // ---- Read all lines from file or stdin ----
336    let lines: Vec<String> = match &args.queue_file {
337        Some(path) if path.to_str() != Some("-") => {
338            eprintln!("  source     : {}", path.display());
339            let file = tokio::fs::File::open(path).await.map_err(|e| {
340                format!("cannot open queue file '{}': {e}", path.display())
341            })?;
342            let mut reader = BufReader::new(file).lines();
343            let mut v = Vec::new();
344            while let Some(line) = reader.next_line().await? {
345                v.push(line);
346            }
347            v
348        }
349        _ => {
350            eprintln!("  source     : stdin");
351            let stdin = tokio::io::stdin();
352            let mut reader = BufReader::new(stdin).lines();
353            let mut v = Vec::new();
354            while let Some(line) = reader.next_line().await? {
355                v.push(line);
356            }
357            v
358        }
359    };
360
361    // ---- Parse entries ----
362    let entries: Vec<ReplayEntry> = lines.iter().filter_map(|l| parse_line(l)).collect();
363    eprintln!("  entries    : {} parsed from {} lines", entries.len(), lines.len());
364
365    if entries.is_empty() {
366        eprintln!("No valid entries to replay. Exiting.");
367        return Ok(());
368    }
369
370    if args.dry_run {
371        eprintln!("\nDry-run mode — no requests submitted.");
372        for (i, entry) in entries.iter().enumerate() {
373            eprintln!(
374                "  [{:>4}] session={} prompt={:?}",
375                i + 1,
376                entry.session_id.as_deref().unwrap_or("-"),
377                entry.prompt.chars().take(60).collect::<String>()
378            );
379        }
380        return Ok(());
381    }
382
383    // ---- Build HTTP client ----
384    let client = Client::builder()
385        .timeout(Duration::from_secs(args.request_timeout_secs))
386        .build()?;
387
388    let api_key = args.api_key.as_deref();
389    let delay = Duration::from_millis(args.delay_ms);
390
391    // ---- Replay loop ----
392    let mut bar = ProgressBar::new(entries.len());
393
394    for entry in &entries {
395        let succeeded = submit_with_retry(
396            &client,
397            &infer_endpoint,
398            entry,
399            api_key,
400            args.max_retries,
401        )
402        .await;
403
404        bar.advance(succeeded, false);
405
406        if delay.as_millis() > 0 {
407            tokio::time::sleep(delay).await;
408        }
409    }
410
411    bar.print_summary();
412
413    // Exit with a non-zero code when any requests failed.
414    if bar.failed > 0 {
415        std::process::exit(1);
416    }
417
418    Ok(())
419}
