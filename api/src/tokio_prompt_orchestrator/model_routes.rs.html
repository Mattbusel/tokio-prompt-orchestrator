1//! Model routing: which upstream answers which model name.
2//!
3//! A static `[[models]]` table maps model names, or globs such as `gpt-*`,
4//! `claude-*` or `llama-*`, to providers. Each route gets its own pipeline
5//! (queue, circuit breaker, timeout, retries, concurrency limit), so a
6//! failing provider only opens its own breaker.
7//!
8//! ```toml
9//! # models.toml, or the same tables inside a pipeline TOML
10//! default_route = true        # unknown models go to --provider/--model (the default)
11//!
12//! [[models]]
13//! name = "gpt-*"              # glob: * is any run of characters, ? one character
14//! provider = "openai"         # any OpenAI-compatible server
15//! # base_url = "https://openrouter.ai/api/v1"
16//! # api_key_env = "OPENROUTER_API_KEY"
17//!
18//! [[models]]
19//! name = "claude-*"
20//! provider = "anthropic"
21//! max_tokens = 8192           # sent when the client gives no max_tokens
22//!
23//! [[models]]
24//! name = "llama-3.1-8b"       # an exact name
25//! provider = "llama_cpp"
26//! base_url = "http://gpu-box:8080"
27//! ```
28//!
29//! Matching is case-insensitive. Exact names are tried first, then globs in
30//! table order. A glob route sends the client's model name upstream; set
31//! `model` on a route to send a fixed name instead.
32//!
33//! With `default_route = true` (the default) a model that matches no route,
34//! or no model at all, goes to the server's own pipeline: the single
35//! `--provider` / `--model` the orchestrator was started with, exactly as
36//! before routing existed. With `default_route = false` such a request is
37//! answered 400 with the list of configured models.
38//!
39//! Load a table with [`ModelRoutesConfig::from_file`] (the CLI's
40//! `--models <file>`) and start its pipelines with [`ModelRoutes::spawn`];
41//! hand the result to the web server through
42//! `web_api::ServerConfig::model_routes`.
43

55/// The kind of upstream a route talks to.
56#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize)]
57#[serde(rename_all = "snake_case")]
58#[non_exhaustive]
59pub enum RouteProvider {
60    /// Any OpenAI-compatible Chat Completions server: OpenAI itself,
61    /// OpenRouter, Groq, Together, vLLM, LM Studio, Ollama's `/v1`.
62    #[serde(rename = "openai")]
63    OpenAi,
64    /// Anthropic's Messages API.
65    Anthropic,
66    /// A llama.cpp server (its OpenAI-compatible `/v1/chat/completions`).
67    LlamaCpp,
68    /// Echo worker: answers with the prompt. For tests and offline setups.
69    Echo,
70}

72impl RouteProvider {
73    /// The provider's name as written in config and reported by `/v1/models`.
74    pub fn as_str(self) -> &'static str {
75        match self {
76            RouteProvider::OpenAi => "openai",
77            RouteProvider::Anthropic => "anthropic",
78            RouteProvider::LlamaCpp => "llama_cpp",
79            RouteProvider::Echo => "echo",
80        }
81    }
82
83    /// `true` for providers that do not bill per token (llama.cpp, echo).
84    pub fn is_free(self) -> bool {
85        matches!(self, RouteProvider::LlamaCpp | RouteProvider::Echo)
86    }
87}

95impl<'de> Deserialize<'de> for RouteProvider {
96    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
97        let s = String::deserialize(deserializer)?;
98        match s.to_ascii_lowercase().replace(['-', '.'], "_").as_str() {
99            "openai" | "openai_compatible" | "open_ai" => Ok(RouteProvider::OpenAi),
100            "anthropic" => Ok(RouteProvider::Anthropic),
101            "llama_cpp" | "llama" | "llamacpp" => Ok(RouteProvider::LlamaCpp),
102            "echo" => Ok(RouteProvider::Echo),
103            other => Err(serde::de::Error::custom(format!(
104                "unknown provider {other:?}; expected one of: openai (any OpenAI-compatible server), anthropic, llama_cpp, echo"
105            ))),
106        }
107    }

110/// One `[[models]]` entry: a model name or glob and the upstream serving it.
111///
112/// # Examples
113///
114/// ```
115/// use tokio_prompt_orchestrator::{ModelRoute, RouteProvider};
116///
117/// let r = ModelRoute::new("gpt-*", RouteProvider::OpenAi)
118///     .with_base_url("http://localhost:8000/v1");
119/// assert!(r.matches("gpt-4o-mini"));
120/// assert!(!r.matches("claude-sonnet-4-5"));
121/// assert_eq!(r.upstream_model("gpt-4o-mini"), "gpt-4o-mini");
122/// ```
123#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
124#[serde(deny_unknown_fields)]
125#[non_exhaustive]
126pub struct ModelRoute {
127    /// Model name, or a glob (`*` any run of characters, `?` one character).
128    pub name: String,
129    /// The kind of upstream.
130    pub provider: RouteProvider,
131    /// Base URL of the upstream. Defaults: `https://api.openai.com/v1`,
132    /// `https://api.anthropic.com/v1`, and `LLAMA_CPP_URL` or
133    /// `http://localhost:8080` for llama.cpp.
134    #[serde(default, skip_serializing_if = "Option::is_none")]
135    pub base_url: Option<String>,
136    /// Environment variable holding the API key. Defaults to
137    /// `OPENAI_API_KEY` for `openai` without a `base_url` and
138    /// `ANTHROPIC_API_KEY` for `anthropic`; an OpenAI-compatible route with
139    /// a `base_url` and no `api_key_env` sends no key (local servers).
140    #[serde(default, skip_serializing_if = "Option::is_none")]
141    pub api_key_env: Option<String>,
142    /// Model name sent upstream instead of the client's.
143    #[serde(default, skip_serializing_if = "Option::is_none")]
144    pub model: Option<String>,
145    /// Reply cap sent when the client sends no `max_tokens`. Anthropic
146    /// requires one, so Anthropic routes default to 4096; other routes send
147    /// none by default and the model's own limit applies.
148    #[serde(default, skip_serializing_if = "Option::is_none")]
149    pub max_tokens: Option<u32>,
150    /// Per-call upstream timeout for this route, in seconds. Defaults to the
151    /// pipeline's, and for Anthropic routes to at least
152    /// [`ANTHROPIC_DEFAULT_TIMEOUT_SECS`] (see
153    /// [`effective_timeout`](Self::effective_timeout)).
154    #[serde(default, skip_serializing_if = "Option::is_none")]
155    pub timeout_secs: Option<u64>,
156}
157
158/// Reply cap for Anthropic routes when neither the client nor the route
159/// sets one (the Messages API requires a value).
160pub const ANTHROPIC_DEFAULT_MAX_TOKENS: u32 = 4096;
161
162/// Shortest per-call timeout an Anthropic route gets when it sets no
163/// `timeout_secs`: Claude answers of [`ANTHROPIC_DEFAULT_MAX_TOKENS`] tokens
164/// often take longer than the 30 s used for other providers. The CLI gives
165/// `--provider anthropic` the same 60 s.
166pub const ANTHROPIC_DEFAULT_TIMEOUT_SECS: u64 = 60;

185        self
186    }

188    /// Read the API key from environment variable `var`.
189    pub fn with_api_key_env(mut self, var: impl Into<String>) -> Self {
190        self.api_key_env = Some(var.into());
191        self
192    }

194    /// Send `model` upstream instead of the client's model name.
195    pub fn with_model(mut self, model: impl Into<String>) -> Self {
196        self.model = Some(model.into());
197        self

200    /// Reply cap used when the client sends none.
201    pub fn with_max_tokens(mut self, max_tokens: u32) -> Self {
202        self.max_tokens = Some(max_tokens);
203        self
204    }

206    /// Per-call upstream timeout for this route, in seconds.
207    pub fn with_timeout_secs(mut self, secs: u64) -> Self {
208        self.timeout_secs = Some(secs);
209        self
210    }
211
212    /// The per-call timeout this route's pipeline uses when the server's own
213    /// pipeline uses `pipeline_default`: the route's `timeout_secs` when set;
214    /// otherwise `pipeline_default`, raised to
215    /// [`ANTHROPIC_DEFAULT_TIMEOUT_SECS`] for an Anthropic route.
216    ///
217    /// # Examples
218    ///
219    /// ```
220    /// use std::time::Duration;
221    /// use tokio_prompt_orchestrator::{ModelRoute, RouteProvider};
222    ///
223    /// let thirty = Duration::from_secs(30);
224    /// let claude = ModelRoute::new("claude-*", RouteProvider::Anthropic);
225    /// assert_eq!(claude.effective_timeout(thirty), Duration::from_secs(60));
226    /// let gpt = ModelRoute::new("gpt-*", RouteProvider::OpenAi);
227    /// assert_eq!(gpt.effective_timeout(thirty), thirty);
228    /// assert_eq!(gpt.with_timeout_secs(90).effective_timeout(thirty), Duration::from_secs(90));
229    /// ```
230    pub fn effective_timeout(&self, pipeline_default: Duration) -> Duration {
231        match (self.timeout_secs, self.provider) {
232            (Some(secs), _) => Duration::from_secs(secs),
233            (None, RouteProvider::Anthropic) => {
234                pipeline_default.max(Duration::from_secs(ANTHROPIC_DEFAULT_TIMEOUT_SECS))
235            }
236            (None, _) => pipeline_default,
237        }
238    }

240    /// `true` when `name` contains `*` or `?`.
241    pub fn is_glob(&self) -> bool {
242        self.name.contains(['*', '?'])
243    }

245    /// `true` when this route serves `model` (case-insensitive).
246    pub fn matches(&self, model: &str) -> bool {
247        glob_match(&self.name.to_ascii_lowercase(), &model.to_ascii_lowercase())
248    }

250    /// The model name sent upstream for a client asking for `requested`.
251    pub fn upstream_model(&self, requested: &str) -> String {
252        self.model.clone().unwrap_or_else(|| requested.to_string())
253    }

255    /// The environment variable the key is read from, if any.
256    pub fn key_variable(&self) -> Option<String> {
257        if let Some(var) = &self.api_key_env {
258            return Some(var.clone()).filter(|v| !v.is_empty());
259        }
260        match self.provider {
261            RouteProvider::OpenAi if self.base_url.is_none() => Some("OPENAI_API_KEY".into()),
262            RouteProvider::Anthropic => Some("ANTHROPIC_API_KEY".into()),
263            _ => None,
264        }
265    }

267    /// Check the entry on its own (name, URL).
268    ///
269    /// # Errors
270    ///
271    /// A description of the first problem found.
272    pub fn validate(&self) -> Result<(), String> {
273        if self.name.trim().is_empty() {
274            return Err("a [[models]] entry has an empty name".into());
275        }
276        if let Some(url) = &self.base_url {
277            if !(url.starts_with("http://") || url.starts_with("https://")) {
278                return Err(format!(
279                    "model route '{}': base_url must start with http:// or https://, got '{url}'",
280                    self.name
281                ));
282            }
283        }
284        if self.timeout_secs == Some(0) {
285            return Err(format!("model route '{}': timeout_secs must be at least 1", self.name));
286        }
287        Ok(())
288    }
289
290    /// Build the worker for this route. `timeout` is the per-call HTTP
291    /// timeout.
292    ///
293    /// # Errors
294    ///
295    /// [`OrchestratorError::ConfigError`] when the route's key variable is
296    /// unset or empty.
297    pub fn build_worker(&self, timeout: Duration) -> Result<Arc<dyn ModelWorker>, OrchestratorError> {
298        let key = match self.key_variable() {
299            Some(var) => Some(
300                std::env::var(&var)
301                    .ok()
302                    .filter(|k| !k.is_empty())
303                    .ok_or_else(|| {
304                        OrchestratorError::ConfigError(format!(
305                            "model route '{}' needs an API key in the environment variable {var}, which is not set",
306                            self.name
307                        ))
308                    })?,
309            ),
310            None => None,
311        };
312        let worker_model = self.model.clone().unwrap_or_else(|| self.name.clone());
313        let base = self.base_url.as_deref().map(|u| u.trim_end_matches('/').to_string());
314        Ok(match self.provider {
315            RouteProvider::OpenAi => {
316                let mut w = OpenAiWorker::with_key(worker_model, key.unwrap_or_default())
317                    .with_timeout(timeout);
318                if let Some(base) = base {
319                    w = w.with_base_url(base);
320                }
321                if let Some(n) = self.max_tokens {
322                    w = w.with_max_tokens(n);
323                }
324                Arc::new(w)
325            }
326            RouteProvider::Anthropic => {
327                let mut w = AnthropicWorker::with_key(worker_model, key.unwrap_or_default())
328                    .with_timeout(timeout)
329                    .with_max_tokens(self.max_tokens.unwrap_or(ANTHROPIC_DEFAULT_MAX_TOKENS));
330                if let Some(base) = base {
331                    w = w.with_base_url(base);
332                }
333                Arc::new(w)
334            }
335            RouteProvider::LlamaCpp => {
336                let mut w = LlamaCppWorker::new().with_timeout(timeout);
337                if let Some(base) = base {
338                    w = w.with_url(base);
339                }
340                if let Some(n) = self.max_tokens.and_then(|n| i32::try_from(n).ok()) {
341                    w = w.with_max_tokens(n);
342                }
343                Arc::new(w)
344            }
345            RouteProvider::Echo => Arc::new(EchoWorker::new()),
346        })
347    }
348}

350/// `true` when `text` matches `pattern` (`*` any run, `?` one character).
351pub fn glob_match(pattern: &str, text: &str) -> bool {
352    let p: Vec<char> = pattern.chars().collect();
353    let t: Vec<char> = text.chars().collect();
354    let (mut pi, mut ti) = (0usize, 0usize);
355    let mut star: Option<(usize, usize)> = None;
356    while ti < t.len() {
357        if pi < p.len() && (p[pi] == '?' || p[pi] == t[ti]) {
358            pi += 1;
359            ti += 1;
360        } else if pi < p.len() && p[pi] == '*' {
361            star = Some((pi, ti));
362            pi += 1;
363        } else if let Some((sp, st)) = star {
364            pi = sp + 1;
365            ti = st + 1;
366            star = Some((sp, st + 1));
367        } else {
368            return false;
369        }
370    }
371    while pi < p.len() && p[pi] == '*' {
372        pi += 1;
373    }
374    pi == p.len()
375}

381/// The `[[models]]` table plus `default_route`, as read from TOML.
382///
383/// # Examples
384///
385/// ```
386/// use tokio_prompt_orchestrator::ModelRoutesConfig;
387///
388/// let cfg = ModelRoutesConfig::from_toml_str(r#"
389///     default_route = false
390///     [[models]]
391///     name = "gpt-*"
392///     provider = "openai"
393///     base_url = "http://localhost:8000/v1"
394/// "#).unwrap();
395/// assert_eq!(cfg.models.len(), 1);
396/// assert!(!cfg.default_route);
397/// ```
398#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
399#[non_exhaustive]
400pub struct ModelRoutesConfig {
401    /// Routes, tried exact names first, then globs in this order.
402    #[serde(default)]
403    pub models: Vec<ModelRoute>,
404    /// Send models that match no route to the server's own pipeline (the
405    /// single `--provider` / `--model`). Default `true`.
406    #[serde(default = "default_true")]
407    pub default_route: bool,
408}
409
410impl Default for ModelRoutesConfig {
411    fn default() -> Self {
412        Self {
413            models: Vec::new(),
414            default_route: true,
415        }
416    }
417}
418
419impl ModelRoutesConfig {
420    /// Routes from `models` with the default route on.
421    pub fn new(models: Vec<ModelRoute>) -> Self {
422        Self {
423            models,
424            default_route: true,
425        }
426    }
427
428    /// Read `[[models]]` and `default_route` from TOML. Every other key is
429    /// ignored, so a pipeline TOML works as well as a file of its own.
430    ///
431    /// # Errors
432    ///
433    /// [`OrchestratorError::ConfigError`] for malformed TOML, an unknown
434    /// field or provider in a `[[models]]` entry, or a failed
435    /// [`validate`](Self::validate).
436    pub fn from_toml_str(content: &str) -> Result<Self, OrchestratorError> {
437        let table: toml::Table = content
438            .parse()
439            .map_err(|e| OrchestratorError::ConfigError(format!("model routes: {e}")))?;
440        Self::from_table(&table)
441    }
442
443    /// Read `[[models]]` and `default_route` from an already parsed table.
444    ///
445    /// # Errors
446    ///
447    /// As [`from_toml_str`](Self::from_toml_str).
448    pub fn from_table(table: &toml::Table) -> Result<Self, OrchestratorError> {
449        let mut own = toml::Table::new();
450        for key in ["models", "default_route"] {
451            if let Some(v) = table.get(key) {
452                own.insert(key.to_string(), v.clone());
453            }
454        }
455        let config: Self = toml::Value::Table(own)
456            .try_into()
457            .map_err(|e| OrchestratorError::ConfigError(format!("model routes: {e}")))?;
458        config.validate().map_err(OrchestratorError::ConfigError)?;
459        Ok(config)
460    }
461
462    /// Read a TOML file (a models file or a pipeline TOML).
463    ///
464    /// # Errors
465    ///
466    /// [`OrchestratorError::ConfigError`] when the file cannot be read or
467    /// [`from_toml_str`](Self::from_toml_str) fails.
468    pub fn from_file(path: &Path) -> Result<Self, OrchestratorError> {
469        let content = std::fs::read_to_string(path).map_err(|e| {
470            OrchestratorError::ConfigError(format!("cannot read {}: {e}", path.display()))
471        })?;
472        Self::from_toml_str(&content).map_err(|e| match e {
473            OrchestratorError::ConfigError(m) => {
474                OrchestratorError::ConfigError(format!("{}: {m}", path.display()))
475            }
476            other => other,
477        })
478    }
479
480    /// The longest per-call timeout among the routes (each as
481    /// [`ModelRoute::effective_timeout`] gives it) and `pipeline_default`,
482    /// the server's own pipeline. Size how long the HTTP handler waits for
483    /// an answer from this, so a route with a long timeout is not cut off
484    /// by the web server first.
485    pub fn longest_timeout(&self, pipeline_default: Duration) -> Duration {
486        self.models
487            .iter()
488            .map(|r| r.effective_timeout(pipeline_default))
489            .fold(pipeline_default, Duration::max)
490    }
491
492    /// Check every entry and that no name appears twice.
493    ///
494    /// # Errors
495    ///
496    /// A description of the first problem found.
497    pub fn validate(&self) -> Result<(), String> {
498        let mut seen = std::collections::HashSet::new();
499        for route in &self.models {
500            route.validate()?;
501            if !seen.insert(route.name.to_ascii_lowercase()) {
502                return Err(format!("model route '{}' is listed twice", route.name));
503            }
504        }
505        Ok(())
506    }
507}

535impl ModelRoutes {
536    /// Build every route's worker and start one pipeline per route with
537    /// `options`. Each route's inference timeout is
538    /// [`ModelRoute::effective_timeout`] of `options.inference_timeout`: its
539    /// own `timeout_secs`, else the pipeline's, raised to 60 s for Anthropic.
540    /// Must be called inside a Tokio runtime.
541    ///
542    /// The web server waits `ServerConfig::timeout_seconds` for each answer;
543    /// size it from [`ModelRoutesConfig::longest_timeout`] and the retries.
544    ///
545    /// # Errors
546    ///
547    /// [`OrchestratorError::ConfigError`] for an invalid table or a missing
548    /// API key. Pipelines already started are shut down.
549    pub fn spawn(config: &ModelRoutesConfig, options: &PipelineOptions) -> Result<Self, OrchestratorError> {
550        config.validate().map_err(OrchestratorError::ConfigError)?;
551        let mut workers = Vec::with_capacity(config.models.len());
552        for route in &config.models {
553            let mut opts = options.clone();
554            opts.inference_timeout = route.effective_timeout(options.inference_timeout);
555            // Same shared retry budget, priced for this route's model. A glob
556            // route without a fixed model is priced by its name pattern,
557            // which as an unknown model is priced high, never at $0.
558            if let Some(b) = &opts.retry.budget {
559                let model = route.model.as_deref().unwrap_or(&route.name);
560                opts.retry.budget = Some(crate::enhanced::RetryBudget::for_model(
561                    b.spend.clone(),
562                    route.provider.as_str(),
563                    model,
564                ));
565            }
566            // The pipeline enforces the timeout (and counts it against the
567            // breaker); the worker's own HTTP timeout is a backstop.
568            let worker = route.build_worker(opts.inference_timeout + Duration::from_secs(1))?;
569            workers.push((route.clone(), worker, opts));
570        }
571        let entries = workers
572            .into_iter()
573            .map(|(route, worker, opts)| RouteEntry {
574                route,
575                handles: spawn_pipeline_with(worker, opts),
576            })
577            .collect();
578        Ok(Self {
579            entries,
580            default_route: config.default_route,
581        })
582    }
583
584    /// Routes served by pipelines you started yourself.
585    pub fn from_pipelines(routes: Vec<(ModelRoute, PipelineHandles)>, default_route: bool) -> Self {
586        Self {
587            entries: routes
588                .into_iter()
589                .map(|(route, handles)| RouteEntry { route, handles })
590                .collect(),
591            default_route,
592        }
593    }
594
595    /// `true` when unmatched models go to the server's own pipeline.
596    pub fn has_default_route(&self) -> bool {
597        self.default_route
598    }
599
600    /// The configured routes, in table order.
601    pub fn routes(&self) -> impl Iterator<Item = &ModelRoute> {
602        self.entries.iter().map(|e| &e.route)
603    }
604
605    /// Number of routes.
606    pub fn len(&self) -> usize {
607        self.entries.len()
608    }

610    /// `true` when there are no routes.
611    pub fn is_empty(&self) -> bool {
612        self.entries.is_empty()
613    }
614
615    /// Index of the route serving `model`: an exact name first, then the
616    /// first glob that matches.
617    pub fn resolve(&self, model: &str) -> Option<usize> {
618        let lower = model.to_ascii_lowercase();
619        self.entries
620            .iter()
621            .position(|e| !e.route.is_glob() && e.route.name.to_ascii_lowercase() == lower)
622            .or_else(|| {
623                self.entries
624                    .iter()
625                    .position(|e| e.route.is_glob() && e.route.matches(model))
626            })
627    }
628
629    /// The route at `index`.
630    pub fn route(&self, index: usize) -> Option<&ModelRoute> {
631        self.entries.get(index).map(|e| &e.route)
632    }
633
634    /// The pipeline serving the route at `index`.
635    pub fn pipeline(&self, index: usize) -> Option<&PipelineHandles> {
636        self.entries.get(index).map(|e| &e.handles)
637    }
638
639    /// Stop every route's pipeline.
640    pub fn shutdown(&self) {
641        for e in &self.entries {
642            e.handles.shutdown();
643        }
644    }
645}
