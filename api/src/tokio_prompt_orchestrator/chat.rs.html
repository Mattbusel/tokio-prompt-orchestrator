1//! Structured chat requests: what a client asked for, carried to the worker.
2//!
3//! [`ModelWorker::infer`](crate::ModelWorker::infer) takes one prompt string,
4//! which is enough for "answer this text" but loses everything else a chat
5//! client sends. [`ChatRequest`] keeps it: the model name, the messages as the
6//! client wrote them (including assistant `tool_calls` and `tool` results),
7//! and the sampling and output settings (`temperature`, `top_p`,
8//! `max_tokens` / `max_completion_tokens`, `stop`, `seed`,
9//! `response_format`, `tools`, `tool_choice`, `parallel_tool_calls`).
10//!
11//! [`ModelWorker::chat`](crate::ModelWorker::chat) answers one with a
12//! [`ChatResponse`], which carries the text, any tool calls the model made,
13//! the finish reason and the provider's token usage. `OpenAiWorker` sends the
14//! request to `/chat/completions` field for field, `AnthropicWorker`
15//! translates it to the Messages API, and `LlamaCppWorker` sends it to
16//! llama.cpp's OpenAI-compatible `/v1/chat/completions`. Workers that do not
17//! override `chat` get a default that flattens the messages into one prompt
18//! and calls `infer`, so the settings are ignored there.
19//!
20//! The OpenAI-compatible HTTP endpoint (`POST /v1/chat/completions`, feature
21//! `web-api`) builds a `ChatRequest` for every call and runs it through the
22//! pipeline, so the stage's circuit breaker, timeout, retries and
23//! concurrency limit apply exactly as they do to plain prompts.
24//!
25//! # Deduplication
26//!
27//! [`ChatRequest::dedup_key`] is a SHA-256 over a canonical encoding of
28//! everything that changes the answer: the route, the model, the messages,
29//! the tools and tool choice, every sampling setting and the response
30//! format. Object keys are sorted first, so two clients that send the same
31//! request with fields in a different order share one answer, while two
32//! requests that differ in any setting never do.
33//! [`ChatRequest::is_cacheable`] says whether a completed answer may be
34//! reused after the call finishes: not when `temperature > 0` without a
35//! `seed`, because the client asked for a sampled answer.
36

45/// One chat completion request, in OpenAI Chat Completions terms.
46///
47/// `messages` holds the client's message objects unchanged (`role`,
48/// `content`, and for tool use `tool_calls`, `tool_call_id`, `name`). All
49/// settings are optional; a worker sends only the ones that are set, and
50/// falls back to its own configured defaults for the rest.
51///
52/// # Examples
53///
54/// ```
55/// use serde_json::json;
56/// use tokio_prompt_orchestrator::ChatRequest;
57///
58/// let mut req = ChatRequest::new("gpt-4o-mini", vec![json!({"role": "user", "content": "Hi"})]);
59/// req.temperature = Some(0.0);
60/// req.max_tokens = Some(64);
61/// assert!(req.is_cacheable());
62/// assert_eq!(req.transcript(), "Hi");
63/// ```
64#[derive(Debug, Clone, Default, PartialEq, Serialize, Deserialize)]
65#[non_exhaustive]
66pub struct ChatRequest {
67    /// Model to call. Empty means "the worker's own model".
68    #[serde(default, skip_serializing_if = "String::is_empty")]
69    pub model: String,
70    /// The conversation, as OpenAI message objects.
71    pub messages: Vec<Value>,
72    /// Sampling temperature.
73    #[serde(default, skip_serializing_if = "Option::is_none")]
74    pub temperature: Option<f64>,
75    /// Nucleus sampling cut-off.
76    #[serde(default, skip_serializing_if = "Option::is_none")]
77    pub top_p: Option<f64>,
78    /// Reply length cap (`max_tokens`).
79    #[serde(default, skip_serializing_if = "Option::is_none")]
80    pub max_tokens: Option<u32>,
81    /// Reply length cap for reasoning models (`max_completion_tokens`).
82    #[serde(default, skip_serializing_if = "Option::is_none")]
83    pub max_completion_tokens: Option<u32>,
84    /// Stop sequence(s): a string or an array of strings.
85    #[serde(default, skip_serializing_if = "Option::is_none")]
86    pub stop: Option<Value>,
87    /// Seed for reproducible sampling.
88    #[serde(default, skip_serializing_if = "Option::is_none")]
89    pub seed: Option<i64>,
90    /// Output format, e.g. `{"type": "json_object"}` or a `json_schema`.
91    #[serde(default, skip_serializing_if = "Option::is_none")]
92    pub response_format: Option<Value>,
93    /// Tool (function) definitions the model may call.
94    #[serde(default, skip_serializing_if = "Option::is_none")]
95    pub tools: Option<Vec<Value>>,
96    /// `"auto"`, `"none"`, `"required"` or `{"type": "function", ...}`.
97    #[serde(default, skip_serializing_if = "Option::is_none")]
98    pub tool_choice: Option<Value>,
99    /// Whether the model may call several tools in one turn.
100    #[serde(default, skip_serializing_if = "Option::is_none")]
101    pub parallel_tool_calls: Option<bool>,
102}
103
104impl ChatRequest {
105    /// A request for `model` with `messages` and no settings.
106    pub fn new(model: impl Into<String>, messages: Vec<Value>) -> Self {
107        Self {
108            model: model.into(),
109            messages,
110            ..Self::default()
111        }
112    }
113
114    /// The reply length cap the client asked for: `max_completion_tokens`,
115    /// else `max_tokens`.
116    pub fn reply_limit(&self) -> Option<u32> {
117        self.max_completion_tokens.or(self.max_tokens)
118    }
119
120    /// `true` when the request defines at least one tool. Without tools,
121    /// `tool_choice` and `parallel_tool_calls` mean nothing and are not sent
122    /// upstream (OpenAI answers 400 for either one without `tools`).
123    pub fn has_tools(&self) -> bool {
124        self.tools.as_ref().is_some_and(|t| !t.is_empty())
125    }
126
127    /// `stop` as a list (a single string becomes a one-element list).
128    pub fn stop_sequences(&self) -> Vec<String> {
129        match &self.stop {
130            Some(Value::String(s)) => vec![s.clone()],
131            Some(Value::Array(items)) => items
132                .iter()
133                .filter_map(|v| v.as_str().map(str::to_string))
134                .collect(),
135            _ => Vec::new(),
136        }
137    }
138
139    /// `true` when a completed answer may be reused for an identical request
140    /// that arrives later. A request with `temperature > 0` and no `seed`
141    /// asks for a fresh sample each time, so it is only shared with
142    /// identical requests that are still in flight. Without a temperature
143    /// (the provider's default) or with a seed, the answer is reusable.
144    pub fn is_cacheable(&self) -> bool {
145        self.seed.is_some() || self.temperature.is_none_or(|t| t <= 0.0)
146    }
147
148    /// The canonical form of everything except the messages, as JSON.
149    fn settings_value(&self, route: &str) -> Value {
150        serde_json::json!({
151            "route": route,
152            "model": self.model,
153            "temperature": self.temperature,
154            "top_p": self.top_p,
155            "max_tokens": self.max_tokens,
156            "max_completion_tokens": self.max_completion_tokens,
157            "stop": self.stop_sequences(),
158            "seed": self.seed,
159            "response_format": self.response_format,
160            "tools": self.tools,
161            "tool_choice": self.tool_choice,
162            "parallel_tool_calls": self.parallel_tool_calls,
163        })
164    }
165
166    /// Deduplication key: SHA-256 (hex) over a canonical encoding of `route`,
167    /// the model, the messages, the tools and tool choice, every sampling
168    /// setting and the response format.
169    ///
170    /// Object keys are sorted at every level, so field order does not matter;
171    /// any difference in content or settings gives a different key. `route`
172    /// names the upstream the request goes to, so the same model name served
173    /// by two routes never shares an answer.
174    ///
175    /// # Examples
176    ///
177    /// ```
178    /// use serde_json::json;
179    /// use tokio_prompt_orchestrator::ChatRequest;
180    ///
181    /// let msgs = vec![json!({"role": "user", "content": "Hi"})];
182    /// let mut a = ChatRequest::new("gpt-4o", msgs.clone());
183    /// let mut b = ChatRequest::new("gpt-4o", msgs);
184    /// assert_eq!(a.dedup_key("default"), b.dedup_key("default"));
185    /// b.temperature = Some(0.9);
186    /// assert_ne!(a.dedup_key("default"), b.dedup_key("default"));
187    /// a.temperature = Some(0.9);
188    /// assert_eq!(a.dedup_key("default"), b.dedup_key("default"));
189    /// ```
190    pub fn dedup_key(&self, route: &str) -> String {
191        let mut text = String::from("chat-v1|");
192        write_canonical(&self.settings_value(route), &mut text);
193        text.push('|');
194        write_canonical(&Value::Array(self.messages.clone()), &mut text);
195        hex::encode(Sha256::digest(text.as_bytes()))
196    }
197
198    /// SHA-256 (hex) over the settings alone (everything in
199    /// [`dedup_key`](Self::dedup_key) except the messages). Requests with
200    /// the same fingerprint differ only in what they say; semantic
201    /// deduplication compares prompts only within one fingerprint.
202    pub fn settings_fingerprint(&self, route: &str) -> String {
203        let mut text = String::from("chat-settings-v1|");
204        write_canonical(&self.settings_value(route), &mut text);
205        hex::encode(Sha256::digest(text.as_bytes()))
206    }
207
208    /// The conversation flattened into one prompt, for workers that take a
209    /// single string. A lone user message is returned exactly as written;
210    /// anything longer becomes a `Role: text` transcript separated by blank
211    /// lines. Only text content is kept.
212    pub fn transcript(&self) -> String {
213        transcript(&self.messages)
214    }
215
216    /// The request as an OpenAI Chat Completions body. `model` fills in an
217    /// empty [`model`](Self::model); `temperature` and `max_tokens` are the
218    /// worker's own defaults, used only when the request leaves them unset.
219    /// Without tools (see [`has_tools`](Self::has_tools)), `tools`,
220    /// `tool_choice` and `parallel_tool_calls` are left out.
221    pub fn to_openai_body(
222        &self,
223        model: &str,
224        temperature: Option<f64>,
225        max_tokens: Option<u32>,
226    ) -> Value {
227        let mut body = serde_json::to_value(self).unwrap_or_else(|_| Value::Object(Default::default()));
228        if let Value::Object(map) = &mut body {
229            if self.model.is_empty() {
230                map.insert("model".into(), Value::String(model.to_string()));
231            }
232            if !self.has_tools() {
233                for key in ["tools", "tool_choice", "parallel_tool_calls"] {
234                    map.remove(key);
235                }
236            }
237            if self.temperature.is_none() {
238                if let Some(t) = temperature {
239                    map.insert("temperature".into(), serde_json::json!(t));
240                }
241            }
242            if self.reply_limit().is_none() {
243                if let Some(n) = max_tokens {
244                    map.insert("max_tokens".into(), serde_json::json!(n));
245                }
246            }
247        }
248        body
249    }
250}

252/// Label used for a message role in a flattened transcript.
253fn role_label(role: &str) -> String {
254    match role {
255        "system" | "developer" => "System".to_string(),
256        "user" => "User".to_string(),
257        "assistant" => "Assistant".to_string(),
258        "tool" | "function" => "Tool".to_string(),
259        other => {
260            let mut chars = other.chars();
261            match chars.next() {
262                Some(first) => first.to_uppercase().chain(chars).collect(),
263                None => "User".to_string(),
264            }
265        }
266    }
267}

269/// The text of one message: a string, or the `text` parts of a content array.
270pub(crate) fn message_text(message: &Value) -> String {
271    match message.get("content") {
272        Some(Value::String(s)) => s.clone(),
273        Some(Value::Array(parts)) => parts
274            .iter()
275            .filter(|p| p.get("type").and_then(Value::as_str) == Some("text"))
276            .filter_map(|p| p.get("text").and_then(Value::as_str))
277            .collect::<Vec<_>>()
278            .join("\n"),
279        _ => String::new(),
280    }
281}

283/// Flatten OpenAI messages into one prompt (see [`ChatRequest::transcript`]).
284pub(crate) fn transcript(messages: &[Value]) -> String {
285    let turns: Vec<(String, String)> = messages
286        .iter()
287        .map(|m| {
288            let role = m.get("role").and_then(Value::as_str).unwrap_or("user");
289            (role_label(role), message_text(m))
290        })
291        .collect();
292    if turns.len() == 1 && turns[0].0 == "User" {
293        return turns[0].1.clone();
294    }
295    turns
296        .iter()
297        .filter(|(_, text)| !text.is_empty())
298        .map(|(label, text)| format!("{label}: {text}"))
299        .collect::<Vec<_>>()
300        .join("\n\n")
301}
302
303/// Write `v` as JSON with object keys sorted at every level.
304fn write_canonical(v: &Value, out: &mut String) {
305    match v {
306        Value::Object(map) => {
307            let sorted: BTreeMap<&String, &Value> = map.iter().collect();
308            out.push('{');
309            for (i, (k, val)) in sorted.into_iter().enumerate() {
310                if i > 0 {
311                    out.push(',');
312                }
313                out.push_str(&Value::String(k.clone()).to_string());
314                out.push(':');
315                write_canonical(val, out);
316            }
317            out.push('}');
318        }
319        Value::Array(items) => {
320            out.push('[');
321            for (i, item) in items.iter().enumerate() {
322                if i > 0 {
323                    out.push(',');
324                }
325                write_canonical(item, out);
326            }
327            out.push(']');
328        }
329        scalar => out.push_str(&scalar.to_string()),
330    }
331}

342/// A worker's answer to a [`ChatRequest`].
343///
344/// # Examples
345///
346/// ```
347/// use tokio_prompt_orchestrator::ChatResponse;
348///
349/// let r = ChatResponse::from_text("Hello");
350/// assert_eq!(r.content.as_deref(), Some("Hello"));
351/// assert_eq!(r.finish_reason.as_deref(), Some("stop"));
352/// assert!(r.tool_calls.is_empty());
353/// ```
354#[derive(Debug, Clone, Default, PartialEq, Serialize, Deserialize)]
355#[non_exhaustive]
356pub struct ChatResponse {
357    /// The reply text; `None` when the model only called tools.
358    #[serde(default)]
359    pub content: Option<String>,
360    /// Tool calls in OpenAI's shape:
361    /// `{"id", "type": "function", "function": {"name", "arguments"}}`.
362    #[serde(default)]
363    pub tool_calls: Vec<Value>,
364    /// `stop`, `length`, `tool_calls` or `content_filter`.
365    #[serde(default)]
366    pub finish_reason: Option<String>,
367    /// Token usage as the provider reported it, if it did.
368    #[serde(default)]
369    pub usage: Option<ChatUsage>,
370}
371
372impl ChatResponse {
373    /// A plain text answer with finish reason `stop`.
374    pub fn from_text(text: impl Into<String>) -> Self {
375        Self {
376            content: Some(text.into()),
377            tool_calls: Vec::new(),
378            finish_reason: Some("stop".to_string()),
379            usage: None,
380        }
381    }
382
383    /// The reply text, or `""` when there is none.
384    pub fn text(&self) -> &str {
385        self.content.as_deref().unwrap_or("")
386    }
387
388    /// Parse an OpenAI Chat Completions response body (first choice).
389    ///
390    /// # Errors
391    ///
392    /// Returns a description of the problem when the body has no choices.
393    pub fn from_openai(body: &Value) -> Result<Self, String> {
394        let choice = body
395            .get("choices")
396            .and_then(Value::as_array)
397            .and_then(|c| c.first())
398            .ok_or_else(|| "no choices in the response".to_string())?;
399        let message = choice.get("message").cloned().unwrap_or(Value::Null);
400        let content = match message.get("content") {
401            Some(Value::String(s)) => Some(s.clone()),
402            Some(Value::Array(_)) => Some(message_text(&message)),
403            _ => None,
404        };
405        let tool_calls = message
406            .get("tool_calls")
407            .and_then(Value::as_array)
408            .cloned()
409            .unwrap_or_default();
410        let finish_reason = choice
411            .get("finish_reason")
412            .and_then(Value::as_str)
413            .map(str::to_string);
414        let usage = body.get("usage").and_then(|u| {
415            Some(ChatUsage {
416                prompt_tokens: u.get("prompt_tokens")?.as_u64()?,
417                completion_tokens: u.get("completion_tokens")?.as_u64()?,
418            })
419        });
420        Ok(Self {
421            content,
422            tool_calls,
423            finish_reason,
424            usage,
425        })
426    }
427}

433/// A chat request travelling through the pipeline under one request id.
434///
435/// The pipeline's stage types carry a prompt string; the structured request
436/// rides alongside, keyed by request id, so the inference stage can call
437/// [`ModelWorker::chat`](crate::ModelWorker::chat) and leave the structured
438/// answer here for the caller.
439pub(crate) struct ChatJob {
440    pub(crate) request: ChatRequest,
441    response: parking_lot::Mutex<Option<ChatResponse>>,
442}
443
444impl ChatJob {
445    pub(crate) fn set_response(&self, response: ChatResponse) {
446        *self.response.lock() = Some(response);
447    }
448}
449
450static CHAT_JOBS: LazyLock<DashMap<String, Arc<ChatJob>>> = LazyLock::new(DashMap::new);
451
452/// The chat job registered for `request_id`, if any.
453pub(crate) fn job(request_id: &str) -> Option<Arc<ChatJob>> {
454    CHAT_JOBS.get(request_id).map(|j| Arc::clone(&j))
455}
456
457/// Registration of a chat job; removed again when dropped.
458#[cfg_attr(not(feature = "web-api"), allow(dead_code))]
459pub(crate) struct ChatJobGuard {
460    request_id: String,
461    job: Arc<ChatJob>,
462}
463
464#[cfg_attr(not(feature = "web-api"), allow(dead_code))]
465impl ChatJobGuard {
466    /// Register `request` under `request_id` until the guard is dropped.
467    pub(crate) fn register(request_id: &str, request: ChatRequest) -> Self {
468        let job = Arc::new(ChatJob {
469            request,
470            response: parking_lot::Mutex::new(None),
471        });
472        CHAT_JOBS.insert(request_id.to_string(), Arc::clone(&job));
473        Self {
474            request_id: request_id.to_string(),
475            job,
476        }
477    }
478
479    /// The structured answer the inference stage left, if it left one.
480    pub(crate) fn take_response(&self) -> Option<ChatResponse> {
481        self.job.response.lock().take()
482    }
483}
484
485impl Drop for ChatJobGuard {
486    fn drop(&mut self) {
487        CHAT_JOBS.remove(&self.request_id);
488    }
489}
