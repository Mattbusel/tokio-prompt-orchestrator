1//! Retry Logic
2//!
3//! Automatic retry with backoff for transient failures, built on the
4//! [`backon`](https://docs.rs/backon) crate. This module keeps the
5//! orchestrator's own policy types ([`RetryPolicy`], [`InferenceRetry`]) and
6//! lets `backon` do the attempt counting, delay schedule, jitter and sleeping.
7//!
8//! ## Usage
9//!
10//! ```no_run
11//! use std::time::Duration;
12//! use tokio_prompt_orchestrator::enhanced::RetryPolicy;
13//! # #[tokio::main]
14//! # async fn main() {
15//! let policy = RetryPolicy::exponential(3, Duration::from_millis(100));
16//!
17//! let result = policy.retry(|| async {
18//!     // Your fallible operation: returns Ok on success, Err on transient failure
19//!     Ok::<String, std::io::Error>("inference result".to_string())
20//! }).await;
21//!
22//! match result {
23//!     Ok(output) => println!("{output}"),
24//!     Err(e) => eprintln!("All retries exhausted: {e}"),
25//! }
26//! # }
27//! ```

38/// Configuration for automatic retry behaviour with backoff.
39///
40/// Construct via the factory helpers [`RetryPolicy::fixed`],
41/// [`RetryPolicy::exponential`], or [`RetryPolicy::linear`], then execute an
42/// async closure with [`RetryPolicy::retry`].
43///
44/// # Examples
45///
46/// ```no_run
47/// use std::time::Duration;
48/// use tokio_prompt_orchestrator::enhanced::RetryPolicy;
49///
50/// # #[tokio::main]
51/// # async fn main() -> Result<(), String> {
52/// let policy = RetryPolicy::exponential(3, Duration::from_millis(100));
53///
54/// let result: Result<String, String> = policy.retry(|| async {
55///     // replace with a real async call
56///     Err("transient".to_string())
57/// }).await;
58///
59/// // All 3 attempts failed
60/// assert!(result.is_err());
61/// # Ok(()) }
62/// ```
63#[derive(Clone, Debug)]
64pub struct RetryPolicy {
65    /// Maximum number of attempts (including the first try).
66    pub max_attempts: usize,
67    /// Backoff strategy applied between consecutive attempts.
68    pub strategy: RetryStrategy,
69}
70
71/// Backoff algorithm applied between successive [`RetryPolicy`] attempts.
72///
73/// - [`RetryStrategy::Fixed`]: constant pause.
74/// - [`RetryStrategy::Exponential`]: the delay is multiplied after each
75///   failure, up to a cap.
76/// - [`RetryStrategy::Linear`]: the delay grows by a fixed increment.
77#[derive(Clone, Debug)]
78pub enum RetryStrategy {
79    /// Fixed delay between retries
80    Fixed(Duration),
81    /// Exponential backoff (delay doubles each time)
82    Exponential {
83        /// Delay applied before the second attempt.
84        initial_delay: Duration,
85        /// Upper bound; the delay will never grow beyond this value.
86        max_delay: Duration,
87        /// Multiplier applied to the delay on each successive failure.
88        multiplier: f64,
89    },
90    /// Linear backoff (delay increases linearly)
91    Linear {
92        /// Delay applied before the second attempt.
93        initial_delay: Duration,
94        /// Amount added to the delay on each successive failure.
95        increment: Duration,
96    },
97}
98
99/// Retry result
100#[derive(Debug)]
101pub enum RetryResult<T, E> {
102    /// Operation succeeded
103    Success(T),
104    /// All retries exhausted
105    Failed {
106        /// The error returned by the final attempt.
107        last_error: E,
108        /// Total number of attempts made before giving up.
109        attempts: usize,
110    },
111}
112
113/// The delay schedule of a [`RetryPolicy`], as a `backon` backoff.
114///
115/// Fixed and exponential schedules are `backon`'s own; linear has no
116/// `backon` builder, so it is a plain iterator.
117enum PolicyBackoff {
118    Fixed(ConstantBackoff),
119    Exponential(ExponentialBackoff),
120    Linear {
121        next: Duration,
122        increment: Duration,
123        remaining: usize,
124    },
125}
126
127impl Iterator for PolicyBackoff {
128    type Item = Duration;
129
130    fn next(&mut self) -> Option<Duration> {
131        match self {
132            Self::Fixed(b) => b.next(),
133            // backon multiplies in f32 (40 ms comes out as 39.999999 ms);
134            // round to whole milliseconds as the policy always has.
135            Self::Exponential(b) => b
136                .next()
137                .map(|d| Duration::from_millis((d.as_secs_f64() * 1000.0).round() as u64)),
138            Self::Linear {
139                next,
140                increment,
141                remaining,
142            } => {
143                if *remaining == 0 {
144                    return None;
145                }
146                *remaining -= 1;
147                let current = *next;
148                *next = next.saturating_add(*increment);
149                Some(current)
150            }
151        }
152    }
153}

155impl RetryPolicy {
156    /// Create a policy that waits a constant `delay` between attempts.
157    ///
158    /// # Arguments
159    ///
160    /// * `max_attempts`: total attempts including the first try. `1` means
161    ///   no retries.
162    /// * `delay`: fixed pause between consecutive attempts.
163    ///
164    /// # Examples
165    ///
166    /// ```
167    /// use std::time::Duration;
168    /// use tokio_prompt_orchestrator::enhanced::RetryPolicy;
169    ///
170    /// let policy = RetryPolicy::fixed(3, Duration::from_millis(50));
171    /// assert_eq!(policy.max_attempts, 3);
172    /// ```
173    pub fn fixed(max_attempts: usize, delay: Duration) -> Self {
174        Self {
175            max_attempts,
176            strategy: RetryStrategy::Fixed(delay),
177        }
178    }
179
180    /// Create a policy with exponential back-off (multiplier 2x, cap 60 s).
181    ///
182    /// The delay before attempt `n + 1` is `min(initial_delay * 2^(n-1), 60 s)`.
183    ///
184    /// # Examples
185    ///
186    /// ```
187    /// use std::time::Duration;
188    /// use tokio_prompt_orchestrator::enhanced::RetryPolicy;
189    ///
190    /// // Delays: 100 ms, 200 ms, 400 ms
191    /// let policy = RetryPolicy::exponential(4, Duration::from_millis(100));
192    /// ```
193    pub fn exponential(max_attempts: usize, initial_delay: Duration) -> Self {
194        Self {
195            max_attempts,
196            strategy: RetryStrategy::Exponential {
197                initial_delay,
198                max_delay: Duration::from_secs(60),
199                multiplier: 2.0,
200            },
201        }
202    }
203
204    /// Create policy with linear backoff
205    pub fn linear(max_attempts: usize, initial_delay: Duration, increment: Duration) -> Self {
206        Self {
207            max_attempts,
208            strategy: RetryStrategy::Linear {
209                initial_delay,
210                increment,
211            },
212        }
213    }
214
215    /// The delays between attempts, `retries` of them at most.
216    fn backoff(&self, retries: usize) -> PolicyBackoff {
217        match &self.strategy {
218            RetryStrategy::Fixed(delay) => PolicyBackoff::Fixed(
219                ConstantBuilder::new()
220                    .with_delay(*delay)
221                    .with_max_times(retries)
222                    .build(),
223            ),
224            RetryStrategy::Exponential {
225                initial_delay,
226                max_delay,
227                multiplier,
228            } => PolicyBackoff::Exponential(
229                ExponentialBuilder::new()
230                    .with_min_delay(*initial_delay)
231                    .with_max_delay(*max_delay)
232                    .with_factor(*multiplier as f32)
233                    .with_max_times(retries)
234                    .build(),
235            ),
236            RetryStrategy::Linear {
237                initial_delay,
238                increment,
239            } => PolicyBackoff::Linear {
240                next: *initial_delay,
241                increment: *increment,
242                remaining: retries,
243            },
244        }
245    }
246
247    /// Delay slept after failed attempt number `attempt` (1-based).
248    #[cfg(test)]
249    fn calculate_delay(&self, attempt: usize) -> Duration {
250        self.backoff(attempt)
251            .nth(attempt.saturating_sub(1))
252            .unwrap_or_default()
253    }
254
255    /// Execute a fallible async closure, retrying on error according to this
256    /// policy.
257    ///
258    /// The closure `f` is called up to `max_attempts` times. Between
259    /// consecutive failures the task sleeps for the delay given by the
260    /// configured [`RetryStrategy`].
261    ///
262    /// # Returns
263    ///
264    /// `Ok(value)` from the first successful attempt, or `Err(last_error)` if
265    /// all attempts fail.
266    ///
267    /// # Examples
268    ///
269    /// ```no_run
270    /// use std::time::Duration;
271    /// use tokio_prompt_orchestrator::enhanced::RetryPolicy;
272    ///
273    /// # #[tokio::main]
274    /// # async fn main() {
275    /// let policy = RetryPolicy::fixed(3, Duration::from_millis(10));
276    /// let result: Result<&str, &str> = policy.retry(|| async { Ok("done") }).await;
277    /// assert_eq!(result, Ok("done"));
278    /// # }
279    /// ```
280    pub async fn retry<F, Fut, T, E>(&self, f: F) -> Result<T, E>
281    where
282        F: FnMut() -> Fut,
283        Fut: std::future::Future<Output = Result<T, E>>,
284        E: std::fmt::Display,
285    {
286        retry_if(self, f, |_| true).await
287    }
288
289    /// Execute with retries, returning detailed result
290    pub async fn retry_with_details<F, Fut, T, E>(&self, f: F) -> RetryResult<T, E>
291    where
292        F: FnMut() -> Fut,
293        Fut: std::future::Future<Output = Result<T, E>>,
294        E: std::fmt::Display,
295    {
296        match self.retry(f).await {
297            Ok(value) => RetryResult::Success(value),
298            Err(error) => RetryResult::Failed {
299                last_error: error,
300                attempts: self.max_attempts,
301            },
302        }
303    }
304
305    /// Check if error is retryable (can be customized)
306    pub fn is_retryable<E>(&self, _error: &E) -> bool {
307        // Default: retry all errors. Use `retry_if` to filter.
308        true
309    }
310}

312/// Retry a fallible async closure only when a predicate approves the error.
313///
314/// Unlike [`RetryPolicy::retry`], errors for which `should_retry` returns
315/// `false` are propagated immediately without consuming further attempts.
316///
317/// # Examples
318///
319/// ```no_run
320/// use std::time::Duration;
321/// use tokio_prompt_orchestrator::enhanced::{RetryPolicy, retry_if};
322///
323/// # #[tokio::main]
324/// # async fn main() {
325/// let policy = RetryPolicy::fixed(5, Duration::from_millis(10));
326/// let result: Result<(), &str> = retry_if(
327///     &policy,
328///     || async { Err("permanent") },
329///     |e| *e == "transient",
330/// ).await;
331/// assert_eq!(result.unwrap_err(), "permanent"); // stopped immediately
332/// # }
333/// ```
334pub async fn retry_if<F, Fut, T, E, P>(policy: &RetryPolicy, f: F, should_retry: P) -> Result<T, E>
335where
336    F: FnMut() -> Fut,
337    Fut: std::future::Future<Output = Result<T, E>>,
338    P: FnMut(&E) -> bool,
339    E: std::fmt::Display,
340{
341    let retries = policy.max_attempts.saturating_sub(1);
342    f.retry(policy.backoff(retries))
343        .when(should_retry)
344        .notify(|e: &E, delay: Duration| {
345            warn!(error = %e, delay_ms = delay.as_millis() as u64, "retry: operation failed, retrying");
346        })
347        .await
348}

350/// Add random jitter (up to 25 %) to a delay to prevent thundering-herd
351/// scenarios when many clients retry at the same time.
352///
353/// # Returns
354///
355/// A `Duration` in the range `[duration, duration + duration/4)`.
356/// When `duration` is very small (< 4 ms), the jitter is zero.
357///
358/// # Examples
359///
360/// ```
361/// use std::time::Duration;
362/// use tokio_prompt_orchestrator::enhanced::retry::with_jitter;
363///
364/// let base = Duration::from_millis(200);
365/// let jittered = with_jitter(base);
366/// assert!(jittered >= base);
367/// assert!(jittered <= base + Duration::from_millis(50));
368/// ```
369pub fn with_jitter(duration: Duration) -> Duration {
370    use rand::Rng;
371    let max_jitter = duration.as_millis() / 4;
372    if max_jitter == 0 {
373        return duration;
374    }
375    let jitter = rand::thread_rng().gen_range(0..max_jitter);
376    duration + Duration::from_millis(jitter as u64)
377}

379/// Execute an inference operation with retries, honouring `RateLimited` back-off.
380///
381/// Unlike `RetryPolicy::retry`, this variant:
382/// - Sleeps for exactly `retry_after_secs` when the provider says to back off.
383/// - Stops at once on errors that cannot succeed on a retry
384///   ([`OrchestratorError::is_retryable`] is `false`: bad key, budget, config).
385/// - Uses jittered exponential backoff (capped at 60 s) for everything else.
386///
387/// ```no_run
388/// use std::time::Duration;
389/// use tokio_prompt_orchestrator::enhanced::retry::retry_inference;
390/// use tokio_prompt_orchestrator::{OrchestratorError, OpenAiWorker, ModelWorker};
391/// # #[tokio::main]
392/// # async fn main() -> Result<(), OrchestratorError> {
393/// # let worker = OpenAiWorker::new("gpt-4o")?;
394/// let tokens = retry_inference(3, Duration::from_millis(200), || async {
395///     worker.infer("hello").await
396/// }).await?;
397/// # Ok(()) }
398/// ```
399pub async fn retry_inference<F, Fut>(
400    max_attempts: usize,
401    base_delay: Duration,
402    f: F,
403) -> Result<Vec<String>, OrchestratorError>
404where
405    F: FnMut() -> Fut,
406    Fut: std::future::Future<Output = Result<Vec<String>, OrchestratorError>>,
407{
408    let backoff = ExponentialBuilder::new()
409        .with_min_delay(base_delay)
410        .with_max_delay(Duration::from_secs(60))
411        .with_max_times(max_attempts.saturating_sub(1))
412        .with_jitter();
413    f.retry(backoff)
414        .when(OrchestratorError::is_retryable)
415        .adjust(|e, delay| match e {
416            OrchestratorError::RateLimited { retry_after_secs } => {
417                delay.map(|_| Duration::from_secs(*retry_after_secs))
418            }
419            _ => delay,
420        })
421        .notify(|e, delay| {
422            warn!(error = %e, delay_ms = delay.as_millis() as u64, "transient error, retrying");
423        })
424        .await
425}

427/// Retry settings for the pipeline's inference stage.
428///
429/// Every model call made by the pipeline goes through
430/// [`InferenceRetry::run`]. With `max_retries == 0` (the default for
431/// [`spawn_pipeline`](crate::spawn_pipeline)) the call is made exactly once.
432/// Otherwise a call that fails with a transient error (429, 5xx, network)
433/// is retried with jittered exponential backoff, and a provider's
434/// `Retry-After` is respected when it is no longer than `max_delay`.
435/// Errors that cannot succeed on a retry (bad key, budget exceeded, bad
436/// config) are returned at once.
437///
438/// All attempts of one request count as one call for the circuit breaker,
439/// and all of them together must fit in the inference timeout.
440///
441/// # Retries cost money, so they are capped in money
442///
443/// A retry count says nothing about what the retries cost: three retries of
444/// a 200,000-token agent context on a frontier model are dollars, and during
445/// an outage every request retries at once. With a [`RetryBudget`] set
446/// ([`with_budget`](Self::with_budget), on by default in the `orchestrator`
447/// binary), each retry must first be paid for from a dollar budget (a floor
448/// plus 10% of the pipeline's real spend per hour, see
449/// [`llm_resilience::RetrySpend`]). A retry that does not fit is not sent:
450/// the request fails with the error it already has. `max_retries` stays as
451/// an upper bound on attempts.
452///
453/// # Examples
454///
455/// ```
456/// use std::time::Duration;
457/// use tokio_prompt_orchestrator::enhanced::InferenceRetry;
458///
459/// let retry = InferenceRetry::new(2, Duration::from_millis(200), Duration::from_secs(5));
460/// assert!(retry.is_enabled());
461/// assert!(!InferenceRetry::disabled().is_enabled());
462/// ```
463#[derive(Debug, Clone)]
464pub struct InferenceRetry {
465    /// Retries after the first attempt. `0` disables retrying.
466    pub max_retries: u32,
467    /// Delay before the first retry; it doubles on each further retry.
468    pub base_delay: Duration,
469    /// Cap on the backoff delay, and on how long a `Retry-After` may ask
470    /// the stage to wait. A longer `Retry-After` fails the request instead.
471    pub max_delay: Duration,
472    /// Dollar budget every retry must fit in. `None` caps retries by count
473    /// only.
474    pub budget: Option<RetryBudget>,
475}
476
477impl PartialEq for InferenceRetry {
478    /// Equal settings; two budgets compare equal when their prices match
479    /// (the shared spend counters are not compared).
480    fn eq(&self, other: &Self) -> bool {
481        self.max_retries == other.max_retries
482            && self.base_delay == other.base_delay
483            && self.max_delay == other.max_delay
484            && match (&self.budget, &other.budget) {
485                (None, None) => true,
486                (Some(a), Some(b)) => {
487                    a.input_usd_per_mtok == b.input_usd_per_mtok
488                        && a.output_usd_per_mtok == b.output_usd_per_mtok
489                }
490                _ => false,
491            }
492    }
493}

495/// Output tokens a retry is assumed to produce when it is priced before it
496/// is sent.
497pub const RETRY_OUTPUT_TOKENS_ESTIMATE: usize = 512;
498
499/// A dollar budget for pipeline retries: the shared
500/// [`llm_resilience::RetrySpend`] plus the model's prices, so a retry can be
501/// priced before it is sent.
502///
503/// A retry is priced at the prompt's input tokens at the input price plus
504/// [`RETRY_OUTPUT_TOKENS_ESTIMATE`] output tokens at the output price.
505/// Successful calls record their own cost as real spend, which grows the
506/// allowance by 10% of it.
507///
508/// ```
509/// use llm_resilience::RetrySpend;
510/// use tokio_prompt_orchestrator::enhanced::RetryBudget;
511///
512/// // $3 / $15 per million tokens; $0.05 floor, nothing spent yet.
513/// let b = RetryBudget::new(RetrySpend::new(0.05), 3.0, 15.0);
514/// // A 10,000-token prompt: $0.03 input + 512 output tokens ($0.00768).
515/// assert!((b.retry_cost(10_000) - 0.03768).abs() < 1e-9);
516/// assert!(b.spend.try_retry(b.retry_cost(10_000)).is_ok());
517/// assert!(b.spend.try_retry(b.retry_cost(10_000)).is_err()); // $0.05 is used up
518/// ```
519#[derive(Debug, Clone)]
520pub struct RetryBudget {
521    /// The shared budget (clones share it).
522    pub spend: llm_resilience::RetrySpend,
523    /// Input price, USD per million tokens.
524    pub input_usd_per_mtok: f64,
525    /// Output price, USD per million tokens.
526    pub output_usd_per_mtok: f64,
527}
528
529impl RetryBudget {
530    /// A budget with explicit prices.
531    pub fn new(
532        spend: llm_resilience::RetrySpend,
533        input_usd_per_mtok: f64,
534        output_usd_per_mtok: f64,
535    ) -> Self {
536        Self {
537            spend,
538            input_usd_per_mtok,
539            output_usd_per_mtok,
540        }
541    }
542
543    /// A budget priced for `provider` / `model`: free for local providers
544    /// (`echo`, `llama`, `llama_cpp`), otherwise the model's list price from
545    /// [`CostEstimator`](crate::cost_estimator::CostEstimator) (an unknown
546    /// model is priced high, never at $0).
547    pub fn for_model(spend: llm_resilience::RetrySpend, provider: &str, model: &str) -> Self {
548        let (input, output) = match provider {
549            "echo" | "llama" | "llama_cpp" => (0.0, 0.0),
550            _ => crate::cost_estimator::CostEstimator::new().prices_per_mtok(model),
551        };
552        Self::new(spend, input, output)
553    }
554
555    /// Cost of a call with these token counts.
556    pub fn call_cost(&self, input_tokens: usize, output_tokens: usize) -> f64 {
557        (input_tokens as f64 * self.input_usd_per_mtok
558            + output_tokens as f64 * self.output_usd_per_mtok)
559            / 1_000_000.0
560    }
561
562    /// What one retry of a `prompt_tokens`-token prompt is priced at.
563    pub fn retry_cost(&self, prompt_tokens: usize) -> f64 {
564        self.call_cost(prompt_tokens, RETRY_OUTPUT_TOKENS_ESTIMATE)
565    }
566}

574impl InferenceRetry {
575    /// No retries: every request makes exactly one model call.
576    pub const fn disabled() -> Self {
577        Self {
578            max_retries: 0,
579            base_delay: Duration::from_millis(100),
580            max_delay: Duration::from_secs(5),
581            budget: None,
582        }

584
585    /// Retry up to `max_retries` times, starting at `base_delay` and capping
586    /// each wait at `max_delay`.
587    pub const fn new(max_retries: u32, base_delay: Duration, max_delay: Duration) -> Self {
588        Self {
589            max_retries,
590            base_delay,
591            max_delay,
592            budget: None,
593        }
594    }
595
596    /// Make every retry pay for itself from `budget` first; a retry that
597    /// does not fit fails the request with the error it already has.
598    #[must_use]
599    pub fn with_budget(mut self, budget: RetryBudget) -> Self {
600        self.budget = Some(budget);
601        self
602    }
603
604    /// Book the cost of a successful call as real spend, which raises the
605    /// retry allowance. No-op without a budget.
606    pub fn record_call(&self, prompt_tokens: usize, output_tokens: usize) {
607        if let Some(b) = &self.budget {
608            b.spend
609                .record_spend(b.call_cost(prompt_tokens, output_tokens));
610        }
611    }
612
613    /// Whether the budget (if any) pays for one more retry of a
614    /// `prompt_tokens`-token prompt. Books the cost when it does.
615    fn pay_for_retry(&self, prompt_tokens: usize) -> bool {
616        let Some(b) = &self.budget else {
617            return true;
618        };
619        match b.spend.try_retry(b.retry_cost(prompt_tokens)) {
620            Ok(()) => true,
621            Err(refused) => {
622                crate::metrics::inc_retry_budget_refused();
623                warn!(
624                    target: "orchestrator::pipeline",
625                    estimated_usd = refused.estimated_usd(),
626                    remaining_usd = refused.remaining_usd(),
627                    allowance_usd = refused.allowance_usd(),
628                    "Retry refused by the dollar retry budget; failing with the original error"
629                );
630                false
631            }
632        }
633    }

639            r.retry_attempts,
640            Duration::from_millis(r.retry_base_ms),
641            Duration::from_millis(r.retry_max_ms),

645    /// `true` when at least one retry is allowed.
646    pub const fn is_enabled(&self) -> bool {

650    /// Run `f`, retrying transient failures according to these settings.
651    /// Retries are priced as a zero-token prompt; use
652    /// [`run_for_prompt`](Self::run_for_prompt) when a budget is set.
653    pub async fn run<F, Fut, T>(&self, f: F) -> Result<T, OrchestratorError>
654    where
655        F: FnMut() -> Fut,
656        Fut: std::future::Future<Output = Result<T, OrchestratorError>>,
657    {
658        self.run_for_prompt(0, f).await
659    }
660
661    /// Run `f`, retrying transient failures according to these settings,
662    /// with each retry priced for a `prompt_tokens`-token prompt and paid
663    /// from the budget (if any) before it is sent.
664    pub async fn run_for_prompt<F, Fut, T>(
665        &self,
666        prompt_tokens: usize,
667        mut f: F,
668    ) -> Result<T, OrchestratorError>
669    where
670        F: FnMut() -> Fut,
671        Fut: std::future::Future<Output = Result<T, OrchestratorError>>,
672    {
673        if !self.is_enabled() {
674            return f().await;
675        }
676        let max_delay = self.max_delay;
677        let backoff = ExponentialBuilder::new()
678            .with_min_delay(self.base_delay.min(max_delay))
679            .with_max_delay(max_delay)
680            .with_max_times(self.max_retries as usize)
681            .with_jitter();
682        f.retry(backoff)
683            .when(|e| e.is_retryable() && self.pay_for_retry(prompt_tokens))
684            .adjust(move |e, delay| match e {
685                // Wait as long as the provider asked, unless that is longer
686                // than we are allowed to wait: then give up and fail fast.
687                OrchestratorError::RateLimited { retry_after_secs } => {
688                    let asked = Duration::from_secs(*retry_after_secs);
689                    if asked > max_delay {
690                        None
691                    } else {
692                        delay.map(|d| d.max(asked))
693                    }
694                }
695                _ => delay,
696            })
697            .notify(|e, delay| {
698                crate::metrics::inc_inference_retry();
699                warn!(
700                    target: "orchestrator::pipeline",
701                    error_kind = e.error_kind(),
702                    delay_ms = delay.as_millis() as u64,
703                    "Inference failed with a transient error, retrying"
704                );
705            })
706            .await
707    }
708}

917    #[tokio::test]
918    async fn test_inference_retry_stops_when_the_dollar_budget_is_spent() {
919        let calls = Arc::new(AtomicUsize::new(0));
920        let c = calls.clone();
921        // $3 / $15 per 1M tokens; a 100,000-token prompt retry costs
922        // $0.30 + 512 * $15 / 1M = $0.30768. A $0.70 floor pays for two.
923        let budget = RetryBudget::new(llm_resilience::RetrySpend::new(0.70), 3.0, 15.0);
924        let retry = InferenceRetry::new(10, Duration::from_millis(1), Duration::from_millis(5))
925            .with_budget(budget.clone());
926        let result: Result<(), _> = retry
927            .run_for_prompt(100_000, || {
928                c.fetch_add(1, Ordering::SeqCst);
929                async { Err(OrchestratorError::Inference("503".into())) }
930            })
931            .await;
932        assert!(matches!(result, Err(OrchestratorError::Inference(_))));
933        assert_eq!(
934            calls.load(Ordering::SeqCst),
935            3,
936            "first try plus the 2 retries $0.70 pays for"
937        );
938        let stats = budget.spend.stats();
939        assert_eq!((stats.granted, stats.refused), (2, 1));
940
941        // A free model's retries cost nothing, so the count is the only cap.
942        let free =
943            InferenceRetry::new(3, Duration::from_millis(1), Duration::from_millis(5)).with_budget(
944                RetryBudget::new(llm_resilience::RetrySpend::new(0.0), 0.0, 0.0),
945            );
946        let calls = Arc::new(AtomicUsize::new(0));
947        let c = calls.clone();
948        let _: Result<(), _> = free
949            .run_for_prompt(100_000, || {
950                c.fetch_add(1, Ordering::SeqCst);
951                async { Err(OrchestratorError::Inference("503".into())) }
952            })
953            .await;
954        assert_eq!(calls.load(Ordering::SeqCst), 4);
955    }
