1//! Semantic Request Deduplication
2//!
3//! Unlike the exact-match [`Deduplicator`] (which only catches byte-identical
4//! requests), `SemanticDeduplicator` uses **SimHash** — a locality-sensitive
5//! hashing technique — to detect *near-duplicate* prompts.  Two prompts that
6//! are paraphrases of the same question produce SimHash fingerprints that
7//! differ in very few bits, and are coalesced to avoid redundant LLM calls.
8//!
9//! ## How SimHash Works
10//!
11//! 1. The input is split into overlapping 3-gram token windows.
12//! 2. Each token is hashed with FNV-1a into a 64-bit value.
13//! 3. For each of the 64 bit positions, we accumulate `+1` if the token's
14//!    hash has that bit set, `−1` otherwise.
15//! 4. The SimHash fingerprint is the sign vector of the 64 accumulators.
16//! 5. Two fingerprints whose **Hamming distance** ≤ `similarity_threshold`
17//!    bits are considered semantically equivalent.
18//!
19//! A threshold of 3 bits catches common paraphrases (punctuation changes,
20//! synonym substitution).  A threshold of 0 bits is exact-content matching.
21//!
22//! ## Example
23//!
24//! ```rust
25//! use tokio_prompt_orchestrator::enhanced::SemanticDeduplicator;
26//! use std::time::Duration;
27//!
28//! let dedup = SemanticDeduplicator::new(3, Duration::from_secs(300));
29//!
30//! // First request is always novel.
31//! assert!(dedup.is_novel("What is the capital of France?"));
32//! dedup.register("What is the capital of France?");
33//!
34//! // Near-duplicate (punctuation change) is caught.
35//! assert!(!dedup.is_novel("What is the capital of France"));
36//!
37//! // Distinct question remains novel.
38//! assert!(dedup.is_novel("What is the capital of Germany?"));
39//! ```
40//!
41//! [`Deduplicator`]: super::Deduplicator

43use dashmap::DashMap;
44use std::sync::Arc;
45use std::time::{Duration, SystemTime};
46use tracing::{debug, trace};

48/// A 64-bit SimHash fingerprint of a text string.
49#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
50pub struct SimHashFingerprint(u64);
51
52impl SimHashFingerprint {
53    /// Compute the SimHash fingerprint for a text string.
54    ///
55    /// The text is tokenised by whitespace, then a 3-gram sliding window is
56    /// applied over the tokens so that small edits produce similar fingerprints
57    /// rather than random ones.
58    pub fn compute(text: &str) -> Self {
59        // Collect tokens, stripping leading/trailing punctuation so that
60        // "France?" and "France" produce the same token hash.
61        let tokens: Vec<String> = text
62            .split_whitespace()
63            .map(|t| t.trim_matches(|c: char| !c.is_alphanumeric()).to_lowercase())
64            .filter(|t| !t.is_empty())
65            .collect();
66
67        // Accumulator: one i32 per bit position.
68        let mut acc = [0i32; 64];
69
70        // Generate weighted shingles using 1-grams and 2-grams.
71        // 1-grams carry weight 1; 2-grams carry weight 2 (they capture context).
72        let process = |token: &str, weight: i32, acc: &mut [i32; 64]| {
73            let h = fnv1a_64(token.as_bytes());
74            for bit in 0..64u32 {
75                if (h >> bit) & 1 == 1 {
76                    acc[bit as usize] += weight;
77                } else {
78                    acc[bit as usize] -= weight;
79                }
80            }
81        };
82
83        for (i, token) in tokens.iter().enumerate() {
84            process(token, 1, &mut acc);
85            if i + 1 < tokens.len() {
86                let bigram = format!("{} {}", token, tokens[i + 1]);
87                process(&bigram, 2, &mut acc);
88            }
89        }
90
91        // Convert accumulator to fingerprint: 1 if acc[i] > 0, else 0.
92        let mut fingerprint = 0u64;
93        for bit in 0..64u32 {
94            if acc[bit as usize] > 0 {
95                fingerprint |= 1u64 << bit;
96            }
97        }
98
99        Self(fingerprint)
100    }
101
102    /// Returns the Hamming distance between two fingerprints (number of bits
103    /// that differ).  Two fingerprints with distance ≤ `threshold` are
104    /// considered semantically similar.
105    #[inline]
106    pub fn hamming_distance(self, other: Self) -> u32 {
107        (self.0 ^ other.0).count_ones()
108    }
109
110    /// Raw fingerprint value.
111    pub fn value(self) -> u64 {
112        self.0
113    }
114}

116/// A cached entry in the semantic dedup store.
117#[derive(Debug, Clone)]
118struct CacheEntry {
119    fingerprint: SimHashFingerprint,
120    inserted_at: SystemTime,
121    /// The canonical key (first text that created this entry) for logging.
122    canonical_key: String,
123}
124
125/// Near-duplicate request detector using SimHash locality-sensitive hashing.
126///
127/// Thread-safe and lock-free on the hot path via `DashMap`.
128///
129/// # Parameters
130///
131/// - `similarity_threshold` — Maximum Hamming distance for two prompts to be
132///   considered duplicates.  Typical values: 0 (exact), 3 (paraphrases),
133///   6 (loose semantic equivalence).
134/// - `window` — How long fingerprints are kept before expiry.
135#[derive(Clone)]
136pub struct SemanticDeduplicator {
137    inner: Arc<SemanticDedupInner>,
138}
139
140struct SemanticDedupInner {
141    /// Stored fingerprints: `fingerprint_value → CacheEntry`.
142    ///
143    /// We store the fingerprint as the key so that lookups are O(1) for exact
144    /// fingerprint matches.  Near-duplicates (different fingerprint, small
145    /// Hamming distance) require a linear scan of the store; this is acceptable
146    /// because the store is bounded by TTL expiry and typical LLM workloads
147    /// have a modest number of distinct in-flight fingerprints.
148    store: DashMap<u64, CacheEntry>,
149    similarity_threshold: u32,
150    window: Duration,
151}
152
153impl SemanticDeduplicator {
154    /// Create a new `SemanticDeduplicator`.
155    ///
156    /// # Arguments
157    /// * `similarity_threshold` — Hamming-distance threshold.  Prompts whose
158    ///   fingerprints differ by ≤ this many bits are treated as duplicates.
159    ///   Use `0` for exact-content dedup only.
160    /// * `window` — TTL for cached fingerprints.
161    pub fn new(similarity_threshold: u32, window: Duration) -> Self {
162        Self {
163            inner: Arc::new(SemanticDedupInner {
164                store: DashMap::new(),
165                similarity_threshold,
166                window,
167            }),
168        }
169    }
170
171    /// Returns `true` if `text` has **not** been seen recently (or is not a
172    /// near-duplicate of a recently seen prompt).
173    ///
174    /// This is a read-only check; call [`register`](Self::register) to record
175    /// a novel request.
176    pub fn is_novel(&self, text: &str) -> bool {
177        let fp = SimHashFingerprint::compute(text);
178        self.find_match(fp).is_none()
179    }
180
181    /// Record `text` as a recently processed request.
182    ///
183    /// Call this after confirming a request is novel (via `is_novel`) and
184    /// sending it to the LLM.
185    pub fn register(&self, text: &str) {
186        let fp = SimHashFingerprint::compute(text);
187        let entry = CacheEntry {
188            fingerprint: fp,
189            inserted_at: SystemTime::now(),
190            canonical_key: text.chars().take(80).collect(),
191        };
192        self.inner.store.insert(fp.value(), entry);
193        trace!(fingerprint = fp.value(), "semantic_dedup: registered fingerprint");
194        self.evict_expired();
195    }
196
197    /// Check if `text` is novel and, if so, register it atomically.
198    ///
199    /// Returns `true` if the request was novel and has been registered.
200    /// Returns `false` if it matched a cached near-duplicate.
201    pub fn check_and_register(&self, text: &str) -> bool {
202        let fp = SimHashFingerprint::compute(text);
203        self.evict_expired();
204        if let Some(existing) = self.find_match(fp) {
205            debug!(
206                fingerprint = fp.value(),
207                matched_canonical = %existing.canonical_key,
208                hamming = fp.hamming_distance(existing.fingerprint),
209                "semantic_dedup: near-duplicate suppressed"
210            );
211            return false;
212        }
213        let entry = CacheEntry {
214            fingerprint: fp,
215            inserted_at: SystemTime::now(),
216            canonical_key: text.chars().take(80).collect(),
217        };
218        self.inner.store.insert(fp.value(), entry);
219        true
220    }
221
222    /// Number of fingerprints currently cached (including possibly expired ones
223    /// not yet evicted).
224    pub fn cached_count(&self) -> usize {
225        self.inner.store.len()
226    }
227
228    /// Manually evict all expired entries.  Called automatically on every
229    /// [`register`](Self::register) and `check_and_register` call.
230    pub fn evict_expired(&self) {
231        let now = SystemTime::now();
232        self.inner.store.retain(|_, entry| {
233            now.duration_since(entry.inserted_at)
234                .map(|age| age < self.inner.window)
235                .unwrap_or(false)
236        });
237    }
238
239    /// Find a cache entry whose fingerprint is within `similarity_threshold`
240    /// Hamming bits of `fp`.  Returns the first match found.
241    fn find_match(&self, fp: SimHashFingerprint) -> Option<CacheEntry> {
242        let now = SystemTime::now();
243        for entry in self.inner.store.iter() {
244            let age = now
245                .duration_since(entry.inserted_at)
246                .unwrap_or(Duration::ZERO);
247            if age >= self.inner.window {
248                continue; // Expired — skip without evicting (done lazily).
249            }
250            if fp.hamming_distance(entry.fingerprint) <= self.inner.similarity_threshold {
251                return Some(entry.clone());
252            }
253        }
254        None
255    }
256}

258/// FNV-1a 64-bit hash — fast, well-distributed, suitable for SimHash tokens.
259#[inline]
260fn fnv1a_64(bytes: &[u8]) -> u64 {
261    const OFFSET_BASIS: u64 = 14_695_981_039_346_656_037;
262    const PRIME: u64 = 1_099_511_628_211;
263    let mut hash = OFFSET_BASIS;
264    for &byte in bytes {
265        hash ^= u64::from(byte);
266        hash = hash.wrapping_mul(PRIME);
267    }
268    hash
269}

271#[cfg(test)]
272mod tests {
273    use super::*;
274
275    #[test]
276    fn identical_texts_produce_same_fingerprint() {
277        let a = SimHashFingerprint::compute("What is the capital of France?");
278        let b = SimHashFingerprint::compute("What is the capital of France?");
279        assert_eq!(a, b);
280        assert_eq!(a.hamming_distance(b), 0);
281    }
282
283    #[test]
284    fn minor_edit_produces_low_hamming_distance() {
285        let a = SimHashFingerprint::compute("What is the capital of France?");
286        let b = SimHashFingerprint::compute("What is the capital of France");
287        // Punctuation drop should produce very few bit flips.
288        assert!(a.hamming_distance(b) <= 5, "distance was {}", a.hamming_distance(b));
289    }
290
291    #[test]
292    fn completely_different_texts_have_high_hamming_distance() {
293        let a = SimHashFingerprint::compute("What is the capital of France?");
294        let b = SimHashFingerprint::compute("How do I bake a chocolate cake?");
295        // Unrelated topics should differ substantially.
296        assert!(a.hamming_distance(b) > 10, "distance was {}", a.hamming_distance(b));
297    }
298
299    #[test]
300    fn is_novel_returns_true_for_unseen_text() {
301        let dedup = SemanticDeduplicator::new(3, Duration::from_secs(300));
302        assert!(dedup.is_novel("Hello, world!"));
303    }
304
305    #[test]
306    fn check_and_register_deduplicates_near_duplicates() {
307        let dedup = SemanticDeduplicator::new(3, Duration::from_secs(300));
308        assert!(dedup.check_and_register("What is the capital of France?"));
309        // Minor punctuation change — should be caught as near-duplicate.
310        assert!(!dedup.check_and_register("What is the capital of France"));
311    }
312
313    #[test]
314    fn distinct_requests_both_pass() {
315        let dedup = SemanticDeduplicator::new(3, Duration::from_secs(300));
316        assert!(dedup.check_and_register("What is the capital of France?"));
317        assert!(dedup.check_and_register("What is the capital of Germany?"));
318    }
319
320    #[test]
321    fn threshold_zero_is_exact_match_only() {
322        let dedup = SemanticDeduplicator::new(0, Duration::from_secs(300));
323        assert!(dedup.check_and_register("Hello world"));
324        // Exact duplicate is blocked.
325        assert!(!dedup.check_and_register("Hello world"));
326        // Different content passes at threshold=0.
327        assert!(dedup.check_and_register("Hello universe"));
328    }
329
330    #[test]
331    fn expired_entries_allow_re_registration() {
332        let dedup = SemanticDeduplicator::new(3, Duration::from_millis(1));
333        assert!(dedup.check_and_register("What is the capital of France?"));
334        // Sleep longer than the TTL.
335        std::thread::sleep(Duration::from_millis(5));
336        // After expiry, the same text should be treated as novel again.
337        assert!(dedup.check_and_register("What is the capital of France?"));
338    }
339
340    #[test]
341    fn empty_text_does_not_panic() {
342        let fp = SimHashFingerprint::compute("");
343        assert_eq!(fp.hamming_distance(fp), 0);
344    }
