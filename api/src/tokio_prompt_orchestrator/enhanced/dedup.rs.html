1//! Request Deduplication
2//!
3//! Prevents duplicate requests from being processed multiple times.
4//! Useful for cost savings when users accidentally submit the same request.
5//!
6//! ## Usage
7//!
8//! ```no_run
9//! use std::time::Duration;
10//! use tokio_prompt_orchestrator::enhanced::{Deduplicator, DeduplicationResult};
11//! # async fn process_request() -> String { String::new() }
12//! # #[tokio::main]
13//! # async fn main() {
14//! let dedup = Deduplicator::new(Duration::from_secs(300)); // 5 minute window
15//!
16//! // Check if request is duplicate
17//! match dedup.check_and_register("prompt_hash").await {
18//!     DeduplicationResult::New(token) => {
19//!         // Process new request
20//!         let result = process_request().await;
21//!         dedup.complete(token, result).await;
22//!     }
23//!     DeduplicationResult::InProgress => {
24//!         // Wait for in-progress request
25//!         let _result = dedup.wait_for_result("prompt_hash").await;
26//!     }
27//!     DeduplicationResult::Cached(result) => {
28//!         // Use cached result
29//!         println!("{result}");
30//!     }
31//! }
32//! # }
33//! ```

47/// Outcome of a [`Deduplicator::check_and_register`] call.
48///
49/// Three-way classification allows callers to decide whether to do the work
50/// themselves, wait for a concurrent worker, or immediately reuse a cached
51/// result.
52///
53/// # Examples
54///
55/// ```no_run
56/// use std::time::Duration;
57/// use tokio_prompt_orchestrator::enhanced::{Deduplicator, DeduplicationResult};
58///
59/// # #[tokio::main]
60/// # async fn main() {
61/// let dedup = Deduplicator::new(Duration::from_secs(60));
62/// match dedup.check_and_register("my-key").await {
63///     DeduplicationResult::New(token) => {
64///         let result = "computed".to_string();
65///         dedup.complete(token, result).await;
66///     }
67///     DeduplicationResult::InProgress => {
68///         // another task is already working; wait for it
69///         let _ = dedup.wait_for_result("my-key").await;
70///     }
71///     DeduplicationResult::Cached(result) => {
72///         println!("reused: {result}");
73///     }
74/// }
75/// # }
76/// ```
77#[derive(Debug, Clone)]
78pub enum DeduplicationResult {
79    /// New request — should be processed by the caller.
80    ///
81    /// The caller must eventually call [`Deduplicator::complete`] or
82    /// [`Deduplicator::fail`] with the returned [`DeduplicationToken`] so that
83    /// any tasks blocked in [`Deduplicator::wait_for_result`] are unblocked.
84    New(DeduplicationToken),
85    /// An identical request is already being processed by another task.
86    ///
87    /// The caller should call [`Deduplicator::wait_for_result`] to block until
88    /// that task completes and then reuse its result.
89    InProgress,
90    /// The request was recently completed and the result is still within the
91    /// cache window.  The cached response string is returned directly.
92    Cached(String),
93}
94
95/// Ownership token issued when a new request is registered with the
96/// [`Deduplicator`].
97///
98/// The holder of a `DeduplicationToken` is the *authoritative worker* for
99/// that request key.  It must call either [`Deduplicator::complete`] or
100/// [`Deduplicator::fail`] to resolve the pending state.
101///
102/// # Drop behaviour
103///
104/// If a token is dropped without calling `complete` or `fail`, the
105/// `InProgress` entry is automatically removed from the deduplicator so that
106/// subsequent callers are not permanently blocked.  A `WARN`-level log line
107/// is emitted in this case.
108///
109/// # Cloning
110///
111/// Tokens are `Clone` because they are cheaply cloneable (`Arc`-backed), but
112/// only the **first** clone to call `complete` or `fail` takes effect;
113/// subsequent calls on other clones are no-ops.
114#[derive(Debug, Clone)]
115pub struct DeduplicationToken {
116    /// Unique identifier for this deduplication token.
117    ///
118    /// Useful for structured log correlation.
119    pub id: String,
120    key: String,
121    completed: Arc<AtomicBool>,
122    /// The same channel waiters subscribed to. Held by the token so that
123    /// waiters are always woken, even if the cache evicted the entry.
124    waiter_tx: broadcast::Sender<String>,
125    requests: Cache<String, RequestState>,
126}
127
128/// # Behavior on Drop
129///
130/// When a `DeduplicationToken` is dropped without calling `complete()`, the
131/// following sequence occurs:
132///
133/// 1. **Cancellation signal**: The `Drop` impl sends a sentinel cancellation
134///    string (`"\x00CANCELLED"`) over the broadcast channel before removing the
135///    entry.  Any tasks already blocked in `wait_for_result()` receive this
136///    value via `rx.recv()` and return `Some("\x00CANCELLED")` rather than
137///    `None`.  Callers of `wait_for_result` that inspect the returned string
138///    can detect cancellation by checking for this sentinel.
139///
140/// 2. **Entry removal**: After the cancellation broadcast the `InProgress`
141///    entry is removed from the shared cache.  Any tasks that subscribe
142///    *after* the removal will find no entry and `wait_for_result` will
143///    return `None`.
144///
145/// 3. **Re-registrability**: Because the entry is removed, the *next* caller
146///    to invoke `check_and_register` for the same key will receive a fresh
147///    `New` token and can retry processing.
148///
149/// **Waiters are NOT left hanging indefinitely.**  They either receive the
150/// cancellation sentinel or `None` (if they race with the removal), both of
151/// which are finite outcomes that unblock the awaiting task promptly.
152///
153/// The sentinel value `"\x00CANCELLED"` uses a NUL prefix which cannot appear
154/// in normal LLM output, making it safe to use as a reserved signal.
155pub const DEDUP_CANCELLED_SENTINEL: &str = "\x00CANCELLED";
156
157/// Returns `true` if the dedup result string represents a cancellation signal.
158///
159/// Callers that receive a result from [`Deduplicator::wait_for_result`] should
160/// use this function instead of comparing to the raw sentinel directly, so
161/// that internal implementation details remain hidden.
162pub fn is_cancelled_result(result: &str) -> bool {
163    result == DEDUP_CANCELLED_SENTINEL
164}

166impl Drop for DeduplicationToken {
167    fn drop(&mut self) {
168        // Only act if this is the last clone and complete() was never called.
169        if Arc::strong_count(&self.completed) == 1 && !self.completed.load(Ordering::Acquire) {
170            // Wake tasks already blocked in wait_for_result() at once.
171            // Send errors only mean nobody is waiting.
172            let _ = self.waiter_tx.send(DEDUP_CANCELLED_SENTINEL.to_string());
173            remove_if_owned_by(&self.requests, &self.key, &self.id);
174            tracing::warn!(
175                key = %self.key,
176                "DeduplicationToken dropped without complete(): cancellation sent and in-progress entry removed"
177            );
178        }
179    }
180}

182/// Deduplicator state
183#[derive(Debug, Clone)]
184enum RequestState {
185    InProgress {
186        /// Id of the [`DeduplicationToken`] that owns this entry.
187        owner: Arc<str>,
188        waiter_tx: broadcast::Sender<String>,
189    },
190    Completed {
191        result: Arc<str>,
192    },
193}

195/// Remove `key` only while it is still the in-progress entry of token `owner`,
196/// so a stale token can never clear a newer registration or a cached result.
197fn remove_if_owned_by(requests: &Cache<String, RequestState>, key: &str, owner: &str) {
198    requests
199        .entry_by_ref(key)
200        .and_compute_with(|current| match current.map(|e| e.into_value()) {
201            Some(RequestState::InProgress { owner: o, .. }) if &*o == owner => Op::Remove,
202            _ => Op::Nop,
203        });
204}

206/// Per-entry lifetimes: a completed result lives for the dedup window; an
207/// in-progress entry lives long enough for a slow model call to finish.
208struct DedupExpiry {
209    cache_duration: Duration,
210    in_progress_ttl: Duration,
211}

213impl Expiry<String, RequestState> for DedupExpiry {
214    fn expire_after_create(
215        &self,
216        _key: &String,
217        value: &RequestState,
218        _created_at: Instant,
219    ) -> Option<Duration> {
220        Some(self.ttl_for(value))
221    }
222
223    fn expire_after_update(
224        &self,
225        _key: &String,
226        value: &RequestState,
227        _updated_at: Instant,
228        _duration_until_expiry: Option<Duration>,
229    ) -> Option<Duration> {
230        Some(self.ttl_for(value))
231    }
232}

234impl DedupExpiry {
235    fn ttl_for(&self, value: &RequestState) -> Duration {
236        match value {
237            RequestState::InProgress { .. } => self.in_progress_ttl,
238            RequestState::Completed { .. } => self.cache_duration,
239        }
240    }
241}

243/// Default bound on tracked keys (in-progress plus cached) per [`Deduplicator`].
244pub const DEFAULT_DEDUP_MAX_ENTRIES: u64 = 100_000;
245
246/// An in-progress entry is dropped after 10x the cache window, but never
247/// sooner than this, so a zero-length window still shares in-flight calls.
248const MIN_IN_PROGRESS_TTL: Duration = Duration::from_secs(600);
249
250/// In-process request deduplicator that coalesces identical concurrent
251/// requests and caches recently completed results.
252///
253/// # How it works
254///
255/// 1. The caller derives a stable cache key (see [`dedup_key`]) from the
256///    prompt and session.
257/// 2. [`Deduplicator::check_and_register`] atomically checks the shared
258///    state and returns one of three outcomes:
259///    - [`DeduplicationResult::New`]: the caller is the first to see this
260///      key; it receives a [`DeduplicationToken`] and must process the request.
261///    - [`DeduplicationResult::InProgress`]: another task is already working;
262///      the caller should call [`Deduplicator::wait_for_result`] to block.
263///    - [`DeduplicationResult::Cached`]: a prior result is still within the
264///      `cache_duration` TTL; the caller can return it immediately.
265/// 3. On success the worker calls [`Deduplicator::complete`]; on failure it
266///    calls [`Deduplicator::fail`], which removes the entry.
267///
268/// # Storage
269///
270/// Entries live in a [`moka`] concurrent cache with a per-entry expiry and a
271/// size bound ([`DEFAULT_DEDUP_MAX_ENTRIES`] unless set with
272/// [`Deduplicator::with_max_entries`]). Expired entries are never returned
273/// and are evicted by the cache itself, so there is no background sweeper
274/// task, and memory stays bounded however many distinct prompts arrive
275/// within one window.
276///
277/// # Thread safety
278///
279/// `Deduplicator` is `Clone + Send + Sync`.  All clones share the same cache.
280///
281/// # Examples
282///
283/// ```no_run
284/// use std::time::Duration;
285/// use tokio_prompt_orchestrator::enhanced::{Deduplicator, DeduplicationResult};
286///
287/// # #[tokio::main]
288/// # async fn main() {
289/// let dedup = Deduplicator::new(Duration::from_secs(300));
290/// let key = "dedup:g:abc123";
291///
292/// match dedup.check_and_register(key).await {
293///     DeduplicationResult::New(token) => {
294///         let result = "hello".to_string();
295///         dedup.complete(token, result).await;
296///     }
297///     DeduplicationResult::InProgress => {
298///         let _ = dedup.wait_for_result(key).await;
299///     }
300///     DeduplicationResult::Cached(result) => println!("{result}"),
301/// }
302/// # }
303/// ```
304#[derive(Clone)]
305pub struct Deduplicator {
306    requests: Cache<String, RequestState>,
307    cache_duration: Duration,
308    /// Optional embedding store for semantic (cosine-similarity) deduplication.
309    embeddings: Arc<DashMap<String, Vec<f32>>>,
310    /// Minimum cosine similarity score to treat a new prompt as a duplicate.
311    similarity_threshold: f32,
312}

314impl Deduplicator {
315    /// Create a new `Deduplicator` with the given cache TTL and room for
316    /// [`DEFAULT_DEDUP_MAX_ENTRIES`] keys.
317    ///
318    /// # Arguments
319    ///
320    /// * `cache_duration`: how long a completed result remains cached before
321    ///   being treated as a fresh request.  Common choices: 5 minutes for
322    ///   interactive use, 1 hour for batch/idempotent workloads. `0` shares
323    ///   only calls that are still in flight.
324    ///
325    /// # Examples
326    ///
327    /// ```
328    /// use std::time::Duration;
329    /// use tokio_prompt_orchestrator::enhanced::Deduplicator;
330    ///
331    /// let dedup = Deduplicator::new(Duration::from_secs(300));
332    /// ```
333    pub fn new(cache_duration: Duration) -> Self {
334        Self::with_max_entries(cache_duration, DEFAULT_DEDUP_MAX_ENTRIES)
335    }
336
337    /// Like [`new`](Self::new), but holding at most `max_entries` keys. When
338    /// full, the cache evicts the entries least likely to be reused.
339    ///
340    /// # Examples
341    ///
342    /// ```
343    /// use std::time::Duration;
344    /// use tokio_prompt_orchestrator::enhanced::Deduplicator;
345    ///
346    /// let dedup = Deduplicator::with_max_entries(Duration::from_secs(60), 10_000);
347    /// ```
348    pub fn with_max_entries(cache_duration: Duration, max_entries: u64) -> Self {
349        let in_progress_ttl = cache_duration.saturating_mul(10).max(MIN_IN_PROGRESS_TTL);
350        let requests = Cache::builder()
351            .max_capacity(max_entries)
352            .expire_after(DedupExpiry {
353                cache_duration,
354                in_progress_ttl,
355            })
356            .build();
357        Self {
358            requests,
359            cache_duration,
360            embeddings: Arc::new(DashMap::new()),
361            similarity_threshold: 1.0, // disabled by default: exact match only
362        }
363    }

365    /// The window a completed result stays reusable for.
366    pub fn cache_duration(&self) -> Duration {
367        self.cache_duration
368    }

370    /// Kept for API compatibility. The cache needs no background task, so
371    /// there is nothing to stop.
372    pub fn signal_shutdown(&self) {}

374    /// Kept for API compatibility. The cache needs no background task, so
375    /// this returns at once.
376    pub async fn shutdown(&self) {}

378    /// Atomically check whether a request is new, in-progress, or cached, and
379    /// register it as in-progress if it is new.
380    ///
381    /// Only one of any number of concurrent callers with the same key
382    /// receives [`DeduplicationResult::New`].
383    ///
384    /// # Arguments
385    ///
386    /// * `key`: stable cache key; derive one with [`dedup_key`].
387    ///
388    /// # Examples
389    ///
390    /// ```no_run
391    /// use std::time::Duration;
392    /// use tokio_prompt_orchestrator::enhanced::{Deduplicator, DeduplicationResult};
393    ///
394    /// # #[tokio::main]
395    /// # async fn main() {
396    /// let dedup = Deduplicator::new(Duration::from_secs(60));
397    /// if let DeduplicationResult::New(token) = dedup.check_and_register("key").await {
398    ///     dedup.complete(token, "result".to_string()).await;
399    /// }
400    /// # }
401    /// ```
402    pub async fn check_and_register(&self, key: &str) -> DeduplicationResult {
403        let id = Uuid::new_v4().to_string();
404        let (tx, _) = broadcast::channel(16);
405        let entry = self
406            .requests
407            .entry_by_ref(key)
408            .or_insert_with(|| RequestState::InProgress {
409                owner: Arc::from(id.as_str()),
410                waiter_tx: tx.clone(),
411            });
412
413        if entry.is_fresh() {
414            let token = DeduplicationToken {
415                id,
416                key: key.to_string(),
417                completed: Arc::new(AtomicBool::new(false)),
418                waiter_tx: tx,
419                requests: self.requests.clone(),
420            };
421            debug!(key = key, token_id = %token.id, "new request registered");
422            return DeduplicationResult::New(token);
423        }
424
425        crate::metrics::inc_dedup_hit();
426        match entry.into_value() {
427            RequestState::InProgress { .. } => {
428                info!(key = key, "duplicate request detected (in progress)");
429                DeduplicationResult::InProgress
430            }
431            RequestState::Completed { result } => {
432                info!(key = key, "duplicate request detected (cached)");
433                DeduplicationResult::Cached(result.to_string())
434            }
435        }
436    }
437
438    /// Wait for an in-progress request to complete and return its result.
439    ///
440    /// If the request has already completed by the time this is called, the
441    /// cached result is returned immediately without waiting.
442    ///
443    /// # Returns
444    ///
445    /// `Some(result)` when the pending request completes, or `None` if the
446    /// key is not tracked (e.g. the worker called [`Deduplicator::fail`]).
447    pub async fn wait_for_result(&self, key: &str) -> Option<String> {
448        let mut rx = match self.requests.get(key)? {
449            RequestState::InProgress { waiter_tx, .. } => waiter_tx.subscribe(),
450            RequestState::Completed { result } => return Some(result.to_string()),
451        };
452
453        let result = rx.recv().await.ok();
454        if result.is_some() {
455            crate::metrics::inc_dedup_waiter_unblocked();
456        }
457        result
458    }
459
460    /// Mark a request as successfully completed and cache its result.
461    ///
462    /// Notifies all tasks currently blocked in [`Deduplicator::wait_for_result`]
463    /// for the same key.  The result is retained for `cache_duration` so
464    /// subsequent callers receive [`DeduplicationResult::Cached`].
465    pub async fn complete(&self, token: DeduplicationToken, result: String) {
466        token.completed.store(true, Ordering::Release);
467        let _ = token.waiter_tx.send(result.clone());
468        if self.cache_duration.is_zero() {
469            // Nothing to cache: just release the key.
470            remove_if_owned_by(&self.requests, &token.key, &token.id);
471        } else {
472            let cached = RequestState::Completed {
473                result: Arc::from(result),
474            };
475            self.requests
476                .entry_by_ref(&token.key)
477                .and_compute_with(|current| match current.map(|e| e.into_value()) {
478                    Some(RequestState::InProgress { owner, .. }) if *owner == *token.id => {
479                        Op::Put(cached)
480                    }
481                    // Evicted while in flight: cache the answer anyway.
482                    None => Op::Put(cached),
483                    _ => Op::Nop,
484                });
485        }
486        info!(key = token.key, token_id = %token.id, "request completed");
487    }
488
489    /// Mark a request as failed and remove it from tracking.
490    ///
491    /// After this call, the next [`Deduplicator::check_and_register`] for the
492    /// same key will receive [`DeduplicationResult::New`] so the request can
493    /// be retried.  Tasks waiting in [`Deduplicator::wait_for_result`]
494    /// receive the cancellation sentinel (see [`is_cancelled_result`]).
495    pub async fn fail(&self, token: DeduplicationToken) {
496        remove_if_owned_by(&self.requests, &token.key, &token.id);
497        debug!(key = token.key, token_id = %token.id, "request failed, removed from dedup");
498    }

500    /// Return a snapshot of current deduplication statistics.
501    ///
502    /// The counts are computed by iterating the cache in O(n). Use sparingly
503    /// on hot paths; prefer Prometheus counters for high-frequency monitoring.
504    pub fn stats(&self) -> DeduplicationStats {
505        let mut stats = DeduplicationStats {
506            total: 0,
507            in_progress: 0,
508            cached: 0,
509        };
510        for (_, state) in self.requests.iter() {
511            stats.total += 1;
512            match state {
513                RequestState::InProgress { .. } => stats.in_progress += 1,
514                RequestState::Completed { .. } => stats.cached += 1,
515            }
516        }
517        stats
518    }

520    /// Clear all cached results
521    pub fn clear(&self) {
522        self.requests.invalidate_all();
523        debug!("deduplication cache cleared");
524    }

526    /// Enable semantic (embedding-based) deduplication.
527    ///
528    /// When enabled, [`check_and_register_with_embedding`](Self::check_and_register_with_embedding)
529    /// compares new embeddings against all stored embeddings using cosine similarity.
530    /// Any stored embedding with similarity >= `threshold` is treated as a cache hit.
531    ///
532    /// # Arguments
533    ///
534    /// * `threshold`: cosine similarity score in `[0.0, 1.0]`.  `1.0` requires
535    ///   exact vector match (default); `0.95` catches near-paraphrases.
536    ///
537    /// # Example
538    ///
539    /// ```
540    /// use std::time::Duration;
541    /// use tokio_prompt_orchestrator::enhanced::Deduplicator;
542    ///
543    /// let dedup = Deduplicator::new(Duration::from_secs(300))
544    ///     .with_semantic(0.95);
545    /// ```
546    pub fn with_semantic(mut self, threshold: f32) -> Self {
547        self.similarity_threshold = threshold;
548        self
549    }
550
551    /// Like [`check_and_register`](Self::check_and_register) but also performs
552    /// a semantic similarity scan against previously registered embeddings.
553    ///
554    /// If `embedding` is `Some` and semantic deduplication is enabled (threshold < 1.0),
555    /// all stored embeddings are scanned.  The first match whose cosine similarity
556    /// meets the threshold is returned as [`DeduplicationResult::Cached`] with an
557    /// empty string (the caller should use `wait_for_result` with the matched key
558    /// to obtain the actual cached value).
559    ///
560    /// Falls back to exact-key lookup when `embedding` is `None` or the threshold
561    /// equals `1.0`.
562    pub async fn check_and_register_with_embedding(
563        &self,
564        key: &str,
565        embedding: Option<Vec<f32>>,
566    ) -> DeduplicationResult {
567        // Semantic scan first (only when an embedding is provided and threshold < 1.0)
568        if let Some(ref emb) = embedding {
569            if self.similarity_threshold < 1.0 {
570                for entry in self.embeddings.iter() {
571                    let sim = cosine_similarity(emb, entry.value());
572                    if sim >= self.similarity_threshold {
573                        debug!(
574                            key = key,
575                            matched_key = entry.key().as_str(),
576                            similarity = sim,
577                            "semantic duplicate detected"
578                        );
579                        crate::metrics::inc_dedup_hit();
580                        return DeduplicationResult::Cached(String::new());
581                    }
582                }
583                // No semantic match: store embedding for future lookups.
584                self.embeddings.insert(key.to_string(), emb.clone());
585            }
586        }
587
588        self.check_and_register(key).await
589    }
590}

592/// Compute the cosine similarity between two dense vectors.
593///
594/// Returns a value in `[-1.0, 1.0]`.  Returns `0.0` if either vector has zero norm
595/// so that zero-length embeddings never falsely match.
596///
597/// # Panics
598///
599/// Does not panic.  Mismatched lengths are handled by iterating the shorter vector.
600pub fn cosine_similarity(a: &[f32], b: &[f32]) -> f32 {
601    let dot: f32 = a.iter().zip(b.iter()).map(|(x, y)| x * y).sum();
602    let norm_a: f32 = a.iter().map(|x| x * x).sum::<f32>().sqrt();
603    let norm_b: f32 = b.iter().map(|x| x * x).sum::<f32>().sqrt();
604    if norm_a == 0.0 || norm_b == 0.0 {
605        0.0
606    } else {
607        dot / (norm_a * norm_b)
608    }
609}

624/// Generate a deduplication key scoped to a session.
625///
626/// Including `session_id` in the key prevents two different sessions from
627/// colliding on the same cached result even when their prompts are identical.
628/// Use `None` only for anonymous/global dedup where cross-session sharing is
629/// intentional (e.g. read-only reference data queries).
630///
631/// # Example
632///
633/// ```
634/// use std::collections::HashMap;
635/// use tokio_prompt_orchestrator::enhanced::dedup_key;
636///
637/// let meta = HashMap::new();
638/// // Same prompt, different sessions → different keys.
639/// let k1 = dedup_key("hello", &meta, Some("session-alice"));
640/// let k2 = dedup_key("hello", &meta, Some("session-bob"));
641/// assert_ne!(k1, k2);
642///
643/// // Same prompt, same session → same key (deterministic).
644/// let k3 = dedup_key("hello", &meta, Some("session-alice"));
645/// assert_eq!(k1, k3);
646///
647/// // No session → global key (backward-compatible).
648/// let k4 = dedup_key("hello", &meta, None);
649/// assert_ne!(k1, k4);
650/// ```
651pub fn dedup_key(
652    prompt: &str,
653    metadata: &std::collections::HashMap<String, String>,
654    session_id: Option<&str>,
655) -> String {
656    // Build a canonical byte sequence: optional session prefix + prompt + sorted metadata.
657    let mut buf = String::new();
658
659    if let Some(sid) = session_id {
660        buf.push_str(sid);
661        buf.push('\x00');
662    }
663
664    buf.push_str(prompt);
665
666    // Include relevant metadata in key (sorted for determinism).
667    let mut meta_keys: Vec<_> = metadata.keys().collect();
668    meta_keys.sort();
669    for key in meta_keys {
670        if let Some(value) = metadata.get(key) {
671            buf.push('\x00');
672            buf.push_str(key);
673            buf.push('=');
674            buf.push_str(value);
675        }
676    }
677
678    // SHA-256, truncated to 128 bits: deterministic across restarts and
679    // processes, and wide enough that two different prompts never share a
680    // cached answer by accident (a 64-bit hash makes that a real risk at scale).
681    let digest = Sha256::digest(buf.as_bytes());
682    let hash = hex::encode(&digest[..16]);
683    match session_id {
684        Some(_) => format!("dedup:s:{hash}"),
685        None => format!("dedup:g:{hash}"),
686    }
687}

783    #[test]
784    fn test_dedup_key_generation() {
785        let empty = HashMap::new();
786
787        // Deterministic: same inputs → same key.
788        let key1 = dedup_key("hello", &empty, None);
789        let key2 = dedup_key("hello", &empty, None);
790        assert_eq!(key1, key2);
791
792        // Different prompt → different key.
793        let key3 = dedup_key("world", &empty, None);
794        assert_ne!(key1, key3);
795
796        // With metadata → different from without.
797        let mut meta = HashMap::new();
798        meta.insert("user".to_string(), "alice".to_string());
799        let key4 = dedup_key("hello", &meta, None);
800        assert_ne!(key1, key4);
801
802        // Global keys carry the "g:" prefix.
803        assert!(key1.starts_with("dedup:g:"), "key={key1}");
804    }

806    #[test]
807    fn test_dedup_key_session_isolation() {
808        let empty = HashMap::new();
809
810        // Same prompt, different sessions → different keys.
811        let k_alice = dedup_key("hello", &empty, Some("session-alice"));
812        let k_bob = dedup_key("hello", &empty, Some("session-bob"));
813        assert_ne!(
814            k_alice, k_bob,
815            "different sessions must not share a dedup key"
816        );
817
818        // Same prompt, same session → same key (deterministic).
819        let k_alice2 = dedup_key("hello", &empty, Some("session-alice"));
820        assert_eq!(k_alice, k_alice2);
821
822        // Session key != global key for the same prompt.
823        let k_global = dedup_key("hello", &empty, None);
824        assert_ne!(k_alice, k_global);
825
826        // Session keys carry the "s:" prefix.
827        assert!(k_alice.starts_with("dedup:s:"), "key={k_alice}");
828        assert!(k_global.starts_with("dedup:g:"), "key={k_global}");
829    }
