1//! Request Deduplication
2//!
3//! Prevents duplicate requests from being processed multiple times.
4//! Useful for cost savings when users accidentally submit the same request.
5//!
6//! ## Usage
7//!
8//! ```no_run
9//! use std::time::Duration;
10//! use tokio_prompt_orchestrator::enhanced::{Deduplicator, DeduplicationResult};
11//! # async fn process_request() -> String { String::new() }
12//! # #[tokio::main]
13//! # async fn main() {
14//! let dedup = Deduplicator::new(Duration::from_secs(300)); // 5 minute window
15//!
16//! // Check if request is duplicate
17//! match dedup.check_and_register("prompt_hash").await {
18//!     DeduplicationResult::New(token) => {
19//!         // Process new request
20//!         let result = process_request().await;
21//!         dedup.complete(token, result).await;
22//!     }
23//!     DeduplicationResult::InProgress => {
24//!         // Wait for in-progress request
25//!         let _result = dedup.wait_for_result("prompt_hash").await;
26//!     }
27//!     DeduplicationResult::Cached(result) => {
28//!         // Use cached result
29//!         println!("{result}");
30//!     }
31//! }
32//! # }
33//! ```

46/// Outcome of a [`Deduplicator::check_and_register`] call.
47///
48/// Three-way classification allows callers to decide whether to do the work
49/// themselves, wait for a concurrent worker, or immediately reuse a cached
50/// result.
51///
52/// # Examples
53///
54/// ```no_run
55/// use std::time::Duration;
56/// use tokio_prompt_orchestrator::enhanced::{Deduplicator, DeduplicationResult};
57///
58/// # #[tokio::main]
59/// # async fn main() {
60/// let dedup = Deduplicator::new(Duration::from_secs(60));
61/// match dedup.check_and_register("my-key").await {
62///     DeduplicationResult::New(token) => {
63///         let result = "computed".to_string();
64///         dedup.complete(token, result).await;
65///     }
66///     DeduplicationResult::InProgress => {
67///         // another task is already working; wait for it
68///         let _ = dedup.wait_for_result("my-key").await;
69///     }
70///     DeduplicationResult::Cached(result) => {
71///         println!("reused: {result}");
72///     }
73/// }
74/// # }
75/// ```
76#[derive(Debug, Clone)]
77pub enum DeduplicationResult {
78    /// New request — should be processed by the caller.
79    ///
80    /// The caller must eventually call [`Deduplicator::complete`] or
81    /// [`Deduplicator::fail`] with the returned [`DeduplicationToken`] so that
82    /// any tasks blocked in [`Deduplicator::wait_for_result`] are unblocked.
83    New(DeduplicationToken),
84    /// An identical request is already being processed by another task.
85    ///
86    /// The caller should call [`Deduplicator::wait_for_result`] to block until
87    /// that task completes and then reuse its result.
88    InProgress,
89    /// The request was recently completed and the result is still within the
90    /// cache window.  The cached response string is returned directly.
91    Cached(String),
92}
93
94/// Ownership token issued when a new request is registered with the
95/// [`Deduplicator`].
96///
97/// The holder of a `DeduplicationToken` is the *authoritative worker* for
98/// that request key.  It must call either [`Deduplicator::complete`] or
99/// [`Deduplicator::fail`] to resolve the pending state.
100///
101/// # Drop behaviour
102///
103/// If a token is dropped without calling `complete` or `fail`, the
104/// `InProgress` entry is automatically removed from the deduplicator so that
105/// subsequent callers are not permanently blocked.  A `WARN`-level log line
106/// is emitted in this case.
107///
108/// # Cloning
109///
110/// Tokens are `Clone` because they are cheaply cloneable (`Arc`-backed), but
111/// only the **first** clone to call `complete` or `fail` takes effect;
112/// subsequent calls on other clones are no-ops.
113#[derive(Debug, Clone)]
114pub struct DeduplicationToken {
115    /// Unique identifier for this deduplication token.
116    ///
117    /// Useful for structured log correlation.
118    pub id: String,
119    key: String,
120    completed: Arc<AtomicBool>,
121    /// The same channel waiters subscribed to. Held by the token so that
122    /// waiters are always woken, even if the cache evicted the entry.
123    waiter_tx: broadcast::Sender<String>,
124    requests: Cache<String, RequestState>,
125}
126
127/// # Behavior on Drop
128///
129/// When a `DeduplicationToken` is dropped without calling `complete()`, the
130/// following sequence occurs:
131///
132/// 1. **Cancellation signal**: The `Drop` impl sends a sentinel cancellation
133///    string (`"\x00CANCELLED"`) over the broadcast channel before removing the
134///    entry.  Any tasks already blocked in `wait_for_result()` receive this
135///    value via `rx.recv()` and return `Some("\x00CANCELLED")` rather than
136///    `None`.  Callers of `wait_for_result` that inspect the returned string
137///    can detect cancellation by checking for this sentinel.
138///
139/// 2. **Entry removal**: After the cancellation broadcast the `InProgress`
140///    entry is removed from the shared cache.  Any tasks that subscribe
141///    *after* the removal will find no entry and `wait_for_result` will
142///    return `None`.
143///
144/// 3. **Re-registrability**: Because the entry is removed, the *next* caller
145///    to invoke `check_and_register` for the same key will receive a fresh
146///    `New` token and can retry processing.
147///
148/// **Waiters are NOT left hanging indefinitely.**  They either receive the
149/// cancellation sentinel or `None` (if they race with the removal), both of
150/// which are finite outcomes that unblock the awaiting task promptly.
151///
152/// The sentinel value `"\x00CANCELLED"` uses a NUL prefix which cannot appear
153/// in normal LLM output, making it safe to use as a reserved signal.
154pub const DEDUP_CANCELLED_SENTINEL: &str = "\x00CANCELLED";
155
156/// Returns `true` if the dedup result string represents a cancellation signal.
157///
158/// Callers that receive a result from [`Deduplicator::wait_for_result`] should
159/// use this function instead of comparing to the raw sentinel directly, so
160/// that internal implementation details remain hidden.
161pub fn is_cancelled_result(result: &str) -> bool {
162    result == DEDUP_CANCELLED_SENTINEL
163}
164
165impl Drop for DeduplicationToken {
166    fn drop(&mut self) {
167        // Only act if this is the last clone and complete() was never called.
168        if Arc::strong_count(&self.completed) == 1 && !self.completed.load(Ordering::Acquire) {
169            // Wake tasks already blocked in wait_for_result() at once.
170            // Send errors only mean nobody is waiting.
171            let _ = self.waiter_tx.send(DEDUP_CANCELLED_SENTINEL.to_string());
172            remove_if_owned_by(&self.requests, &self.key, &self.id);
173            tracing::warn!(
174                key = %self.key,
175                "DeduplicationToken dropped without complete(): cancellation sent and in-progress entry removed"
176            );
177        }
178    }
179}

181/// Deduplicator state
182#[derive(Debug, Clone)]
183enum RequestState {
184    InProgress {
185        /// Id of the [`DeduplicationToken`] that owns this entry.
186        owner: Arc<str>,
187        waiter_tx: broadcast::Sender<String>,
188    },
189    Completed {
190        result: Arc<str>,
191    },
192}

194/// Remove `key` only while it is still the in-progress entry of token `owner`,
195/// so a stale token can never clear a newer registration or a cached result.
196fn remove_if_owned_by(requests: &Cache<String, RequestState>, key: &str, owner: &str) {
197    requests
198        .entry_by_ref(key)
199        .and_compute_with(|current| match current.map(|e| e.into_value()) {
200            Some(RequestState::InProgress { owner: o, .. }) if &*o == owner => Op::Remove,
201            _ => Op::Nop,
202        });
203}

205/// Per-entry lifetimes: a completed result lives for the dedup window; an
206/// in-progress entry lives long enough for a slow model call to finish.
207struct DedupExpiry {
208    cache_duration: Duration,
209    in_progress_ttl: Duration,
210}

212impl Expiry<String, RequestState> for DedupExpiry {
213    fn expire_after_create(
214        &self,
215        _key: &String,
216        value: &RequestState,
217        _created_at: Instant,
218    ) -> Option<Duration> {
219        Some(self.ttl_for(value))

223        &self,
224        _key: &String,
225        value: &RequestState,
226        _updated_at: Instant,
227        _duration_until_expiry: Option<Duration>,
228    ) -> Option<Duration> {
229        Some(self.ttl_for(value))
230    }

233impl DedupExpiry {
234    fn ttl_for(&self, value: &RequestState) -> Duration {
235        match value {
236            RequestState::InProgress { .. } => self.in_progress_ttl,
237            RequestState::Completed { .. } => self.cache_duration,
238        }
239    }

277/// In-process request deduplicator that coalesces identical concurrent
278/// requests and caches recently completed results.
279///
280/// # How it works
281///
282/// 1. The caller derives a stable cache key (see [`dedup_key`]) from the
283///    prompt and session.
284/// 2. [`Deduplicator::check_and_register`] atomically checks the shared
285///    state and returns one of three outcomes:
286///    - [`DeduplicationResult::New`]: the caller is the first to see this
287///      key; it receives a [`DeduplicationToken`] and must process the request.
288///    - [`DeduplicationResult::InProgress`]: another task is already working;
289///      the caller should call [`Deduplicator::wait_for_result`] to block.
290///    - [`DeduplicationResult::Cached`]: a prior result is still within the
291///      `cache_duration` TTL; the caller can return it immediately.
292/// 3. On success the worker calls [`Deduplicator::complete`]; on failure it
293///    calls [`Deduplicator::fail`], which removes the entry.
294///
295/// # Storage
296///
297/// Entries live in a [`moka`] concurrent cache with a per-entry expiry and a
298/// size bound ([`DEFAULT_DEDUP_MAX_ENTRIES`] unless set with
299/// [`Deduplicator::with_max_entries`]). Expired entries are never returned
300/// and are evicted by the cache itself, so there is no background sweeper
301/// task, and memory stays bounded however many distinct prompts arrive
302/// within one window.
303///
304/// # Thread safety
305///
306/// `Deduplicator` is `Clone + Send + Sync`.  All clones share the same cache.
307///
308/// # Examples
309///
310/// ```no_run
311/// use std::time::Duration;
312/// use tokio_prompt_orchestrator::enhanced::{Deduplicator, DeduplicationResult};
313///
314/// # #[tokio::main]
315/// # async fn main() {
316/// let dedup = Deduplicator::new(Duration::from_secs(300));
317/// let key = "dedup:g:abc123";
318///
319/// match dedup.check_and_register(key).await {
320///     DeduplicationResult::New(token) => {
321///         let result = "hello".to_string();
322///         dedup.complete(token, result).await;
323///     }
324///     DeduplicationResult::InProgress => {
325///         let _ = dedup.wait_for_result(key).await;
326///     }
327///     DeduplicationResult::Cached(result) => println!("{result}"),
328/// }
329/// # }
330/// ```
331#[derive(Clone)]
332pub struct Deduplicator {
333    requests: Cache<String, RequestState>,
334    cache_duration: Duration,
335    /// Prompt embeddings by dedup key, for semantic matching. Bounded and
336    /// expiring with the answer cache, so it cannot grow without limit.
337    embeddings: Cache<String, Arc<IndexedPrompt>>,
338    /// Minimum cosine similarity score to treat a new prompt as a duplicate.
339    similarity_threshold: f32,
340    /// Set by [`Deduplicator::with_embedder`].
341    semantic: Option<SemanticConfig>,
342}

344impl Deduplicator {
345    /// Create a new `Deduplicator` with the given cache TTL and room for
346    /// [`DEFAULT_DEDUP_MAX_ENTRIES`] keys.
347    ///
348    /// # Arguments
349    ///
350    /// * `cache_duration`: how long a completed result remains cached before
351    ///   being treated as a fresh request.  Common choices: 5 minutes for
352    ///   interactive use, 1 hour for batch/idempotent workloads. `0` shares
353    ///   only calls that are still in flight.
354    ///
355    /// # Examples
356    ///
357    /// ```
358    /// use std::time::Duration;
359    /// use tokio_prompt_orchestrator::enhanced::Deduplicator;
360    ///
361    /// let dedup = Deduplicator::new(Duration::from_secs(300));
362    /// ```
363    pub fn new(cache_duration: Duration) -> Self {
364        Self::with_max_entries(cache_duration, DEFAULT_DEDUP_MAX_ENTRIES)
365    }
366
367    /// Like [`new`](Self::new), but holding at most `max_entries` keys. When
368    /// full, the cache evicts the entries least likely to be reused.
369    ///
370    /// # Examples
371    ///
372    /// ```
373    /// use std::time::Duration;
374    /// use tokio_prompt_orchestrator::enhanced::Deduplicator;
375    ///
376    /// let dedup = Deduplicator::with_max_entries(Duration::from_secs(60), 10_000);
377    /// ```
378    pub fn with_max_entries(cache_duration: Duration, max_entries: u64) -> Self {
379        let in_progress_ttl = cache_duration.saturating_mul(10).max(MIN_IN_PROGRESS_TTL);
380        let requests = Cache::builder()
381            .max_capacity(max_entries)
382            .expire_after(DedupExpiry {
383                cache_duration,
384                in_progress_ttl,
385            })
386            .build();
387        Self {
388            requests,
389            cache_duration,
390            embeddings: Cache::builder()
391                .max_capacity(DEFAULT_SEMANTIC_MAX_ENTRIES)
392                .time_to_live(cache_duration.max(Duration::from_millis(1)))
393                .build(),
394            similarity_threshold: 1.0, // disabled by default: exact match only
395            semantic: None,
396        }
397    }

400    pub fn cache_duration(&self) -> Duration {
401        self.cache_duration
402    }

412    /// Atomically check whether a request is new, in-progress, or cached, and
413    /// register it as in-progress if it is new.
414    ///
415    /// Only one of any number of concurrent callers with the same key
416    /// receives [`DeduplicationResult::New`].
417    ///
418    /// # Arguments
419    ///
420    /// * `key`: stable cache key; derive one with [`dedup_key`].
421    ///
422    /// # Examples
423    ///
424    /// ```no_run
425    /// use std::time::Duration;
426    /// use tokio_prompt_orchestrator::enhanced::{Deduplicator, DeduplicationResult};
427    ///
428    /// # #[tokio::main]
429    /// # async fn main() {
430    /// let dedup = Deduplicator::new(Duration::from_secs(60));
431    /// if let DeduplicationResult::New(token) = dedup.check_and_register("key").await {
432    ///     dedup.complete(token, "result".to_string()).await;
433    /// }
434    /// # }
435    /// ```
436    pub async fn check_and_register(&self, key: &str) -> DeduplicationResult {
437        let id = Uuid::new_v4().to_string();
438        let (tx, _) = broadcast::channel(16);
439        let entry = self
440            .requests
441            .entry_by_ref(key)
442            .or_insert_with(|| RequestState::InProgress {
443                owner: Arc::from(id.as_str()),
444                waiter_tx: tx.clone(),
445            });
446
447        if entry.is_fresh() {
448            let token = DeduplicationToken {
449                id,
450                key: key.to_string(),
451                completed: Arc::new(AtomicBool::new(false)),
452                waiter_tx: tx,
453                requests: self.requests.clone(),
454            };
455            debug!(key = key, token_id = %token.id, "new request registered");
456            return DeduplicationResult::New(token);
457        }
458
459        crate::metrics::inc_dedup_hit();
460        match entry.into_value() {
461            RequestState::InProgress { .. } => {
462                info!(key = key, "duplicate request detected (in progress)");
463                DeduplicationResult::InProgress
464            }
465            RequestState::Completed { result } => {
466                info!(key = key, "duplicate request detected (cached)");
467                DeduplicationResult::Cached(result.to_string())
468            }
469        }
470    }
471
472    /// Wait for an in-progress request to complete and return its result.
473    ///
474    /// If the request has already completed by the time this is called, the
475    /// cached result is returned immediately without waiting.
476    ///
477    /// # Returns
478    ///
479    /// `Some(result)` when the pending request completes, or `None` if the
480    /// key is not tracked (e.g. the worker called [`Deduplicator::fail`]).
481    pub async fn wait_for_result(&self, key: &str) -> Option<String> {
482        let mut rx = match self.requests.get(key)? {
483            RequestState::InProgress { waiter_tx, .. } => waiter_tx.subscribe(),
484            RequestState::Completed { result } => return Some(result.to_string()),
485        };
486
487        let result = rx.recv().await.ok();
488        if result.is_some() {
489            crate::metrics::inc_dedup_waiter_unblocked();
490        }
491        result
492    }
493
494    /// Mark a request as successfully completed and cache its result.
495    ///
496    /// Notifies all tasks currently blocked in [`Deduplicator::wait_for_result`]
497    /// for the same key.  The result is retained for `cache_duration` so
498    /// subsequent callers receive [`DeduplicationResult::Cached`].
499    pub async fn complete(&self, token: DeduplicationToken, result: String) {
500        token.completed.store(true, Ordering::Release);
501        let _ = token.waiter_tx.send(result.clone());
502        if self.cache_duration.is_zero() {
503            // Nothing to cache: just release the key.
504            remove_if_owned_by(&self.requests, &token.key, &token.id);
505        } else {
506            let cached = RequestState::Completed {
507                result: Arc::from(result),
508            };
509            self.requests
510                .entry_by_ref(&token.key)
511                .and_compute_with(|current| match current.map(|e| e.into_value()) {
512                    Some(RequestState::InProgress { owner, .. }) if *owner == *token.id => {
513                        Op::Put(cached)
514                    }
515                    // Evicted while in flight: cache the answer anyway.
516                    None => Op::Put(cached),
517                    _ => Op::Nop,
518                });
519        }
520        info!(key = token.key, token_id = %token.id, "request completed");
521    }
522
523    /// Mark a request as failed and remove it from tracking.
524    ///
525    /// After this call, the next [`Deduplicator::check_and_register`] for the
526    /// same key will receive [`DeduplicationResult::New`] so the request can
527    /// be retried.  Tasks waiting in [`Deduplicator::wait_for_result`]
528    /// receive the cancellation sentinel (see [`is_cancelled_result`]).
529    pub async fn fail(&self, token: DeduplicationToken) {
530        remove_if_owned_by(&self.requests, &token.key, &token.id);
531        debug!(key = token.key, token_id = %token.id, "request failed, removed from dedup");
532    }

534    /// Return a snapshot of current deduplication statistics.
535    ///
536    /// The counts are computed by iterating the cache in O(n). Use sparingly
537    /// on hot paths; prefer Prometheus counters for high-frequency monitoring.
538    pub fn stats(&self) -> DeduplicationStats {
539        let mut stats = DeduplicationStats {
540            total: 0,
541            in_progress: 0,
542            cached: 0,
543        };
544        for (_, state) in self.requests.iter() {
545            stats.total += 1;
546            match state {
547                RequestState::InProgress { .. } => stats.in_progress += 1,
548                RequestState::Completed { .. } => stats.cached += 1,
549            }
550        }
551        stats
552    }

556        self.requests.invalidate_all();

560    /// Enable semantic (embedding-based) deduplication.
561    ///
562    /// When enabled, [`check_and_register_with_embedding`](Self::check_and_register_with_embedding)
563    /// compares new embeddings against all stored embeddings using cosine similarity.
564    /// Any stored embedding with similarity >= `threshold` is treated as a cache hit.
565    ///
566    /// # Arguments
567    ///
568    /// * `threshold`: cosine similarity score in `[0.0, 1.0]`.  `1.0` requires
569    ///   exact vector match (default); `0.95` catches near-paraphrases.
570    ///
571    /// # Example
572    ///
573    /// ```
574    /// use std::time::Duration;
575    /// use tokio_prompt_orchestrator::enhanced::Deduplicator;
576    ///
577    /// let dedup = Deduplicator::new(Duration::from_secs(300))
578    ///     .with_semantic(0.95);
579    /// ```
580    pub fn with_semantic(mut self, threshold: f32) -> Self {
581        self.similarity_threshold = threshold;
582        self
583    }
584
585    /// Like [`check_and_register`](Self::check_and_register), but a new
586    /// prompt whose `embedding` is close enough (see
587    /// [`with_semantic`](Self::with_semantic)) to an earlier prompt that has
588    /// already been answered gets that answer as
589    /// [`DeduplicationResult::Cached`].
590    ///
591    /// Only completed answers are reused; a similar prompt that is still in
592    /// flight does not count. Falls back to exact-key lookup when `embedding`
593    /// is `None` or the threshold is `1.0`.
594    pub async fn check_and_register_with_embedding(
595        &self,
596        key: &str,
597        embedding: Option<Vec<f32>>,
598    ) -> DeduplicationResult {
599        let threshold = self.similarity_threshold;
600        match embedding {
601            Some(embedding) if threshold < 1.0 => {
602                self.register_semantic(key, embedding, threshold).await.0
603            }
604            _ => self.check_and_register(key).await,
605        }
606    }

608    /// Turn on semantic deduplication with `embedder`: a prompt that means the
609    /// same as one answered within the cache window is answered from the
610    /// cache, even when the wording differs.
611    ///
612    /// `threshold` is the minimum cosine similarity between the two prompts'
613    /// embeddings, and a match must also pass [`same_specifics`] (same
614    /// numbers, shared words in the same order), because embeddings alone
615    /// confuse questions like "convert 10 miles to km" and "convert 10 km to
616    /// miles".
617    ///
618    /// Measured with BGE-small-en-v1.5 (`FastEmbedder::try_default`) in
619    /// `tests/semantic_dedup_tests.rs`:
620    ///
621    /// | Pair | Similarity | Reused at 0.93 |
622    /// |---|---|---|
623    /// | "What is the capital of France?" / "Which city is the capital of France?" | 0.959 | yes |
624    /// | "What time zone is Tokyo in?" / "Which time zone does Tokyo use?" | 0.963 | yes |
625    /// | "How many ounces are in a pound?" / "How many oz in one lb?" | 0.931 | yes |
626    /// | "How do I reverse a list in Python?" / "What's the way to reverse a Python list?" | 0.985 | no (word order) |
627    /// | "Convert 10 miles to kilometers" / "Convert 10 kilometers to miles" | 0.992 | no (guard) |
628    /// | "Is 17 a prime number?" / "Is 21 a prime number?" | 0.845 | no |
629    /// | "What is the capital of France?" / "What is the capital of Germany?" | 0.795 | no |
630    ///
631    /// Other models need their own threshold: measure a few of your own
632    /// pairs the same way before relying on it.
633    ///
634    /// Use [`check_and_register_semantic`](Self::check_and_register_semantic)
635    /// to dedup with it; [`check_and_register`](Self::check_and_register)
636    /// stays exact-match only.
637    #[must_use]
638    pub fn with_embedder(
639        mut self,
640        embedder: Arc<dyn crate::embedding::Embedder>,
641        threshold: f32,
642    ) -> Self {
643        self.semantic = Some(SemanticConfig {
644            embedder,
645            threshold: threshold.clamp(-1.0, 1.0),
646        });
647        self
648    }
649
650    /// `true` when [`with_embedder`](Self::with_embedder) was used.
651    pub fn is_semantic(&self) -> bool {
652        self.semantic.is_some()
653    }
654
655    /// Exact dedup first, then (with [`with_embedder`](Self::with_embedder))
656    /// a search for an earlier, already answered prompt with the same
657    /// meaning.
658    ///
659    /// Returns the usual [`DeduplicationResult`] plus, for a semantic hit,
660    /// which prompt matched and how closely. A semantic hit is also cached
661    /// under `key`, so repeating this exact prompt later is an exact hit.
662    ///
663    /// If the embedder fails, the error is logged and the request is treated
664    /// as new: a broken embedder costs a model call, never an answer.
665    pub async fn check_and_register_semantic(
666        &self,
667        key: &str,
668        text: &str,
669    ) -> (DeduplicationResult, Option<SemanticMatch>) {
670        let Some(semantic) = self.semantic.clone() else {
671            return (self.check_and_register(key).await, None);
672        };
673        // An exact hit, or an identical call in flight, needs no embedding.
674        let exact = self.check_and_register(key).await;
675        let DeduplicationResult::New(token) = exact else {
676            return (exact, None);
677        };
678        let embedding = match semantic.embedder.embed(text).await {
679            Ok(embedding) => embedding,
680            Err(e) => {
681                warn!(key = key, error = %e, "embedding failed; treating the request as new");
682                return (DeduplicationResult::New(token), None);
683            }
684        };
685        let found = self.best_completed_match(key, &embedding, Some(text), semantic.threshold);
686        self.embeddings.insert(
687            key.to_string(),
688            Arc::new(IndexedPrompt { embedding, text: Some(text.to_string()) }),
689        );
690        match found {
691            Some((matched, result)) => {
692                crate::metrics::inc_dedup_hit();
693                info!(
694                    key = key,
695                    matched_key = matched.matched_key.as_str(),
696                    similarity = matched.similarity,
697                    "semantic duplicate answered from cache"
698                );
699                self.complete(token, result.clone()).await;
700                (DeduplicationResult::Cached(result), Some(matched))
701            }
702            None => (DeduplicationResult::New(token), None),
703        }
704    }
705
706    /// Exact check, then a similarity search, for a caller-supplied embedding.
707    async fn register_semantic(
708        &self,
709        key: &str,
710        embedding: Vec<f32>,
711        threshold: f32,
712    ) -> (DeduplicationResult, Option<SemanticMatch>) {
713        let exact = self.check_and_register(key).await;
714        let DeduplicationResult::New(token) = exact else {
715            return (exact, None);
716        };
717        let found = self.best_completed_match(key, &embedding, None, threshold);
718        self.embeddings.insert(key.to_string(), Arc::new(IndexedPrompt { embedding, text: None }));
719        match found {
720            Some((matched, result)) => {
721                crate::metrics::inc_dedup_hit();
722                self.complete(token, result.clone()).await;
723                (DeduplicationResult::Cached(result), Some(matched))
724            }
725            None => (DeduplicationResult::New(token), None),
726        }
727    }
728
729    /// The most similar earlier prompt at or above `threshold` whose answer
730    /// is still cached, with that answer.
731    fn best_completed_match(
732        &self,
733        key: &str,
734        embedding: &[f32],
735        text: Option<&str>,
736        threshold: f32,
737    ) -> Option<(SemanticMatch, String)> {
738        let mut best: Option<(SemanticMatch, String)> = None;
739        for (other_key, other) in self.embeddings.iter() {
740            if other_key.as_str() == key {
741                continue;
742            }
743            let similarity = cosine_similarity(embedding, &other.embedding);
744            if similarity < threshold
745                || best.as_ref().is_some_and(|(b, _)| b.similarity >= similarity)
746            {
747                continue;
748            }
749            // Embeddings barely see numbers and word order ("10 miles to km"
750            // vs "10 km to miles" score 0.99 with BGE-small), so a close
751            // vector is not enough: the specifics must agree too.
752            if let (Some(a), Some(b)) = (text, other.text.as_deref()) {
753                if !same_specifics(a, b) {
754                    continue;
755                }
756            }
757            if let Some(RequestState::Completed { result }) = self.requests.get(other_key.as_str())
758            {
759                best = Some((
760                    SemanticMatch {
761                        matched_key: other_key.to_string(),
762                        similarity,
763                    },
764                    result.to_string(),
765                ));
766            }
767        }
768        best
769    }
770}

772/// Words that carry no specifics; ignored by [`same_specifics`].
773const GUARD_STOPWORDS: &[&str] = &[
774    "a", "an", "the", "of", "in", "on", "at", "to", "for", "from", "by", "with", "and", "or",
775    "is", "are", "was", "were", "be", "been", "do", "does", "did", "can", "could", "would",
776    "should", "will", "what", "which", "who", "whom", "whose", "when", "where", "why", "how",
777    "i", "me", "my", "we", "our", "you", "your", "it", "its", "this", "that", "these", "those",
778    "please", "tell", "give", "show", "way", "there", "s", "whats", "what's",
779];
780
781fn guard_tokens(text: &str) -> Vec<String> {
782    text.split(|c: char| !c.is_alphanumeric() && c != '.')
783        .map(|t| t.trim_matches('.').to_lowercase())
784        .filter(|t| !t.is_empty())
785        .collect()
786}
787
788/// `true` when two prompts agree on their specifics, the things embeddings
789/// are known to blur: every number is the same, in the same order, and the
790/// content words they share appear in the same order.
791///
792/// So "Convert 10 miles to kilometers" and "Convert 10 kilometers to miles"
793/// disagree (shared words in a different order), as do "Is 17 prime?" and
794/// "Is 71 prime?" (different numbers), while "What is the capital of
795/// France?" and "Which city is the capital of France?" agree. It errs on the
796/// side of "different": a wrong "different" costs one model call, a wrong
797/// "same" returns someone else's answer.
798pub fn same_specifics(a: &str, b: &str) -> bool {
799    let (ta, tb) = (guard_tokens(a), guard_tokens(b));
800    let is_number = |t: &String| t.chars().any(|c| c.is_ascii_digit());
801    let numbers_a: Vec<&String> = ta.iter().filter(|t| is_number(t)).collect();
802    let numbers_b: Vec<&String> = tb.iter().filter(|t| is_number(t)).collect();
803    if numbers_a != numbers_b {
804        return false;
805    }
806    let content = |tokens: &[String]| -> Vec<String> {
807        tokens
808            .iter()
809            .filter(|t| !is_number(t) && !GUARD_STOPWORDS.contains(&t.as_str()))
810            .cloned()
811            .collect()
812    };
813    let (ca, cb) = (content(&ta), content(&tb));
814    let shared_a: Vec<&String> = ca.iter().filter(|t| cb.contains(t)).collect();
815    let shared_b: Vec<&String> = cb.iter().filter(|t| ca.contains(t)).collect();
816    // Repeated words make the order ambiguous; compare first occurrences.
817    let first = |v: Vec<&String>| -> Vec<String> {
818        let mut seen: Vec<String> = Vec::new();
819        for t in v {
820            if !seen.contains(t) {
821                seen.push(t.clone());
822            }
823        }
824        seen
825    };
826    first(shared_a) == first(shared_b)
827}

844/// Generate a deduplication key scoped to a session.
845///
846/// Including `session_id` in the key prevents two different sessions from
847/// colliding on the same cached result even when their prompts are identical.
848/// Use `None` only for anonymous/global dedup where cross-session sharing is
849/// intentional (e.g. read-only reference data queries).
850///
851/// # Example
852///
853/// ```
854/// use std::collections::HashMap;
855/// use tokio_prompt_orchestrator::enhanced::dedup_key;
856///
857/// let meta = HashMap::new();
858/// // Same prompt, different sessions → different keys.
859/// let k1 = dedup_key("hello", &meta, Some("session-alice"));
860/// let k2 = dedup_key("hello", &meta, Some("session-bob"));
861/// assert_ne!(k1, k2);
862///
863/// // Same prompt, same session → same key (deterministic).
864/// let k3 = dedup_key("hello", &meta, Some("session-alice"));
865/// assert_eq!(k1, k3);
866///
867/// // No session → global key (backward-compatible).
868/// let k4 = dedup_key("hello", &meta, None);
869/// assert_ne!(k1, k4);
870/// ```
871pub fn dedup_key(
872    prompt: &str,
873    metadata: &std::collections::HashMap<String, String>,
874    session_id: Option<&str>,
875) -> String {
876    // Build a canonical byte sequence: optional session prefix + prompt + sorted metadata.
877    let mut buf = String::new();
878
879    if let Some(sid) = session_id {
880        buf.push_str(sid);
881        buf.push('\x00');
882    }
883
884    buf.push_str(prompt);
885
886    // Include relevant metadata in key (sorted for determinism).
887    let mut meta_keys: Vec<_> = metadata.keys().collect();
888    meta_keys.sort();
889    for key in meta_keys {
890        if let Some(value) = metadata.get(key) {
891            buf.push('\x00');
892            buf.push_str(key);
893            buf.push('=');
894            buf.push_str(value);
895        }
896    }
897
898    // SHA-256, truncated to 128 bits: deterministic across restarts and
899    // processes, and wide enough that two different prompts never share a
900    // cached answer by accident (a 64-bit hash makes that a real risk at scale).
901    let digest = Sha256::digest(buf.as_bytes());
902    let hash = hex::encode(&digest[..16]);
903    match session_id {
904        Some(_) => format!("dedup:s:{hash}"),
905        None => format!("dedup:g:{hash}"),
906    }
907}
