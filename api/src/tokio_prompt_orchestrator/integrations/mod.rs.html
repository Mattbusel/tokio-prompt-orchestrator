1//! Use the orchestrator with the Rust LLM crates you already have.
2//!
3//! Each integration is its own cargo feature, off by default, so you only
4//! compile the one you use. They all produce a [`ModelWorker`], so the
5//! deduplication, circuit breaker, retries, timeouts and dead-letter queue
6//! work the same whichever client sits underneath.
7//!
8//! | Feature | Type | Wraps |
9//! |---|---|---|
10//! | `async-openai` | [`AsyncOpenAiWorker`] | an [`async_openai::Client`] you configured: OpenAI, Azure OpenAI, or any OpenAI-compatible server |
11//! | `genai` | [`GenaiWorker`] | a [`genai::Client`]: OpenAI, Anthropic, Gemini, Ollama, Groq, DeepSeek, xAI, Cohere and more, picked by model name |
12//! | `rig` | [`RigWorker`] | any [rig](https://docs.rs/rig-core) agent or model that implements `rig::completion::Prompt` (needs Rust 1.95) |
13//! | `tower` | [`ServiceWorker`], [`WorkerService`] | any `tower::Service<String>`, in both directions |
14//!
15//! A [`Retriever`](crate::Retriever) that grounds prompts in your documents:
16//!
17//! | Feature | Type | Searches |
18//! |---|---|---|
19//! | `tantivy` | [`TantivyRetriever`] | a folder of Markdown/text files, BM25 full-text, in memory (needs Rust 1.90) |
20//!
21//! And [`Embedder`](crate::Embedder)s for semantic deduplication:
22//!
23//! | Feature | Type | Runs |
24//! |---|---|---|
25//! | `fastembed` | `FastEmbedder` | local ONNX models on the CPU, no API key (needs Rust 1.88) |
26//! | `async-openai` | [`OpenAiEmbedder`] | the OpenAI embeddings API or a compatible server |
27//! | `genai` | [`GenaiEmbedder`] | any provider genai supports embeddings for |
28//!
29//! [`ModelWorker`]: crate::ModelWorker

72/// Map an HTTP status and message from a provider client to the error the
73/// pipeline understands: 401/403 are never retried, 429 backs off, and
74/// everything else counts as a (retryable) inference failure.
75#[allow(dead_code)] // unused when only the tower/rig integrations are enabled
76pub(crate) fn error_from_status(
77    status: u16,
78    retry_after_secs: Option<u64>,
79    message: impl std::fmt::Display,
80) -> crate::OrchestratorError {
81    match status {
82        401 | 403 => crate::OrchestratorError::AuthFailed(format!("HTTP {status}: {message}")),
83        429 => crate::OrchestratorError::RateLimited {
84            retry_after_secs: retry_after_secs.unwrap_or(1),
85        },
86        _ => crate::OrchestratorError::Inference(format!("HTTP {status}: {message}")),
87    }
88}

90/// Turn a complete reply into the token list workers return: the whole text
91/// as one entry, or nothing when the model answered with only whitespace
92/// (the same convention as the built-in workers).
93#[allow(dead_code)]
94pub(crate) fn reply_tokens(text: String) -> Vec<String> {
95    if text.trim().is_empty() {
96        Vec::new()
97    } else {
98        vec![text]
99    }
100}

102#[cfg(test)]
103mod tests {
104    use super::*;
105    use crate::OrchestratorError;
106
107    #[test]
108    fn auth_errors_are_not_retryable() {
109        let e = error_from_status(401, None, "bad key");
110        assert!(matches!(e, OrchestratorError::AuthFailed(_)));
111        assert!(!e.is_retryable());
112        assert!(matches!(
113            error_from_status(403, None, "forbidden"),
114            OrchestratorError::AuthFailed(_)
115        ));
116    }
117
118    #[test]
119    fn rate_limits_keep_the_retry_after() {
120        match error_from_status(429, Some(7), "slow down") {
121            OrchestratorError::RateLimited { retry_after_secs } => assert_eq!(retry_after_secs, 7),
122            other => panic!("expected RateLimited, got {other:?}"),
123        }
124    }
125
126    #[test]
127    fn server_errors_are_retryable_inference_failures() {
128        let e = error_from_status(503, None, "overloaded");
129        assert!(matches!(e, OrchestratorError::Inference(_)));
130        assert!(e.is_retryable());
131    }
132
133    #[test]
134    fn whitespace_replies_become_empty() {
135        assert!(reply_tokens("  \n".into()).is_empty());
136        assert_eq!(reply_tokens("Paris".into()), vec!["Paris".to_string()]);
137    }
