1//! Prompt complexity scoring.
2//!
3//! Analyses a prompt string and produces a complexity score in the range
4//! `0.0..=1.0`.  The score drives the routing decision:
5//!
6//! | Score       | Route                                       |
7//! |-------------|---------------------------------------------|
8//! | `< 0.4`     | Local model (free, fast)                    |
9//! | `0.4  -  0.7` | Local model with cloud fallback on failure  |
10//! | `> 0.7`     | Cloud model directly                        |
11//!
12//! ## Heuristics
13//!
14//! 1. **Token count**  -  >500 whitespace-delimited tokens → +0.3
15//! 2. **Code blocks**  -  presence of fenced code blocks (` ``` `) → +0.2
16//! 3. **Multi-step instructions**  -  numbered list items → +0.2
17//! 4. **Ambiguous references**  -  words like "it", "that", "the thing" → +0.15
18//! 5. **Domain-specific terms**  -  Rust/financial keywords → +0.15
19//!
20//! The raw sum is clamped to `[0.0, 1.0]`.
21
22/// A prompt complexity scorer.
23///
24/// Stateless and cheap to construct.  All analysis runs in O(n) over
25/// the prompt length with no heap allocations beyond the input scan.
26///
27/// # Panics
28///
29/// This type and its methods never panic.
30#[derive(Debug, Clone)]
31pub struct ComplexityScorer {
32    /// Minimum token count before the "long prompt" signal fires.
33    token_count_threshold: usize,
34    /// Domain-specific terms that trigger the domain-term complexity signal.
35    /// Loaded from [`super::config::RoutingConfig::domain_terms`] when
36    /// constructed via [`ComplexityScorer::from_config`].
37    domain_terms: Vec<String>,
38}
39
40impl ComplexityScorer {
41    /// Create a new scorer with the default token-count threshold (500) and
42    /// the built-in domain term list.
43    pub fn new() -> Self {
44        Self {
45            token_count_threshold: 500,
46            domain_terms: super::config::RoutingConfig::default().domain_terms,
47        }
48    }
49
50    /// Create a new scorer with a custom token-count threshold and the built-in
51    /// domain term list.
52    ///
53    /// # Arguments
54    ///
55    /// * `token_count_threshold`  -  Prompts with more whitespace-delimited
56    ///   tokens than this value receive the long-prompt bonus.
57    pub fn with_token_threshold(token_count_threshold: usize) -> Self {
58        Self {
59            token_count_threshold,
60            domain_terms: super::config::RoutingConfig::default().domain_terms,
61        }
62    }
63
64    /// Create a scorer configured from a [`super::config::RoutingConfig`].
65    ///
66    /// The domain term list is taken directly from `config.domain_terms` so
67    /// operators can customise scoring without code changes.
68    ///
69    /// # Panics
70    ///
71    /// This function never panics.
72    pub fn from_config(config: &super::config::RoutingConfig) -> Self {
73        Self {
74            token_count_threshold: 500,
75            domain_terms: config.domain_terms.clone(),
76        }
77    }
78
79    /// Score a prompt for complexity.
80    ///
81    /// # Arguments
82    ///
83    /// * `prompt`  -  The raw prompt text to analyse.
84    ///
85    /// # Returns
86    ///
87    /// A `f64` in `[0.0, 1.0]` representing estimated complexity.
88    ///
89    /// # Panics
90    ///
91    /// This function never panics.
92    ///
93    /// # Example
94    ///
95    /// ```rust
96    /// use tokio_prompt_orchestrator::routing::ComplexityScorer;
97    /// let scorer = ComplexityScorer::new();
98    /// let score = scorer.score("Say hello");
99    /// assert!(score < 0.4);
100    /// ```
101    pub fn score(&self, prompt: &str) -> f64 {
102        let mut total = 0.0_f64;
103
104        total += self.token_count_signal(prompt);
105        total += Self::code_block_signal(prompt);
106        total += Self::multi_step_signal(prompt);
107        total += Self::ambiguous_reference_signal(prompt);
108        total += self.domain_term_signal(prompt);
109
110        clamp_score(total)
111    }
112
113    /// Provide a breakdown of individual signal contributions.
114    ///
115    /// Useful for debugging, logging, and transparency into routing decisions.
116    ///
117    /// # Arguments
118    ///
119    /// * `prompt`  -  The raw prompt text to analyse.
120    ///
121    /// # Returns
122    ///
123    /// A [`ScoreBreakdown`] showing each signal's contribution and the
124    /// clamped total.
125    ///
126    /// # Panics
127    ///
128    /// This function never panics.
129    pub fn breakdown(&self, prompt: &str) -> ScoreBreakdown {
130        let token_count = self.token_count_signal(prompt);
131        let code_blocks = Self::code_block_signal(prompt);
132        let multi_step = Self::multi_step_signal(prompt);
133        let ambiguous_refs = Self::ambiguous_reference_signal(prompt);
134        let domain_terms = self.domain_term_signal(prompt);
135        let total =
136            clamp_score(token_count + code_blocks + multi_step + ambiguous_refs + domain_terms);
137
138        ScoreBreakdown {
139            token_count,
140            code_blocks,
141            multi_step,
142            ambiguous_refs,
143            domain_terms,
144            total,
145        }
146    }
147
148    //  Individual signals
149
150    /// +0.3 if the prompt exceeds the token-count threshold.
151    fn token_count_signal(&self, prompt: &str) -> f64 {
152        let count = prompt.split_whitespace().count();
153        if count > self.token_count_threshold {
154            0.3
155        } else {
156            0.0
157        }
158    }
159
160    /// +0.2 if the prompt contains fenced code blocks.
161    fn code_block_signal(prompt: &str) -> f64 {
162        if prompt.contains("```") {
163            0.2
164        } else {
165            0.0
166        }
167    }
168
169    /// +0.2 if the prompt contains numbered list items (multi-step instructions).
170    fn multi_step_signal(prompt: &str) -> f64 {
171        let mut numbered_items = 0_usize;
172        for line in prompt.lines() {
173            let trimmed = line.trim_start();
174            // Match patterns like "1.", "2.", "10." at line start
175            if let Some(rest) = trimmed.strip_suffix('.') {
176                // We want "N." at the start, so check the prefix
177                let _ = rest; // suppress unused
178            }
179            // More robust: check if trimmed starts with digit(s) followed by '.'
180            let mut chars = trimmed.chars();
181            if let Some(first) = chars.next() {
182                if first.is_ascii_digit() {
183                    // Consume remaining digits
184                    let rest: String = chars.collect();
185                    if rest.starts_with('.') || rest.starts_with(|c: char| c.is_ascii_digit()) {
186                        // Check for "N." or "NN." pattern
187                        let dot_pos = trimmed.find('.');
188                        if let Some(pos) = dot_pos {
189                            let prefix = &trimmed[..pos];
190                            if !prefix.is_empty() && prefix.chars().all(|c| c.is_ascii_digit()) {
191                                numbered_items += 1;
192                            }
193                        }
194                    }
195                }
196            }
197        }
198        // Two or more numbered items → multi-step
199        if numbered_items >= 2 {
200            0.2
201        } else {
202            0.0
203        }
204    }
205
206    /// +0.15 if the prompt contains ambiguous pronominal references.
207    fn ambiguous_reference_signal(prompt: &str) -> f64 {
208        let lower = prompt.to_lowercase();
209        let ambiguous_patterns = [
210            " it ",
211            " that ",
212            " the thing ",
213            " this thing ",
214            " those ",
215            " these ",
216            " them ",
217        ];
218        let count = ambiguous_patterns
219            .iter()
220            .filter(|pat| lower.contains(*pat))
221            .count();
222        // Need at least 2 distinct ambiguous patterns to trigger
223        if count >= 2 {
224            0.15
225        } else {
226            0.0
227        }
228    }
229
230    /// +0.15 if the prompt contains domain-specific terminology from
231    /// `self.domain_terms` (loaded from [`super::config::RoutingConfig`]).
232    fn domain_term_signal(&self, prompt: &str) -> f64 {
233        let lower = prompt.to_lowercase();
234        let hits = self
235            .domain_terms
236            .iter()
237            .filter(|t| lower.contains(t.as_str()))
238            .count();
239        // Two or more domain terms triggers the signal
240        if hits >= 2 {
241            0.15
242        } else {
243            0.0
244        }
245    }
246}

248impl Default for ComplexityScorer {
249    fn default() -> Self {
250        Self::new()
251    }

254/// Clamp a raw score to the valid `[0.0, 1.0]` range.
255fn clamp_score(raw: f64) -> f64 {
256    raw.clamp(0.0, 1.0)
257}

259/// Breakdown of individual complexity signal contributions.
260///
261/// Returned by [`ComplexityScorer::breakdown`] for observability.
262///
263/// # Panics
264///
265/// This type never panics.
266#[derive(Debug, Clone, PartialEq)]
267pub struct ScoreBreakdown {
268    /// Contribution from the token-count heuristic (0.0 or 0.3).
269    pub token_count: f64,
270    /// Contribution from code-block detection (0.0 or 0.2).
271    pub code_blocks: f64,
272    /// Contribution from multi-step instruction detection (0.0 or 0.2).
273    pub multi_step: f64,
274    /// Contribution from ambiguous-reference detection (0.0 or 0.15).
275    pub ambiguous_refs: f64,
276    /// Contribution from domain-specific terminology (0.0 or 0.15).
277    pub domain_terms: f64,
278    /// Final clamped score in `[0.0, 1.0]`.
279    pub total: f64,
280}
