83/// Configuration for the model routing layer.
84///
85/// Controls how the [`super::ComplexityScorer`] output maps to a
86/// [`super::RoutingDecision`] and how per-token costs are tracked.
87///
88/// # Panics
89///
90/// This type never panics.
91#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
92pub struct RoutingConfig {
93    /// Complexity score below which requests route to the local model.
94    ///
95    /// Range: `0.0..=1.0`.  Default: `0.4`.
96    #[serde(default = "default_local_threshold")]
97    pub local_threshold: f64,
98
99    /// Complexity score above which requests route directly to the cloud model.
100    ///
101    /// Range: `0.0..=1.0`, must be `>= local_threshold`.  Default: `0.7`.
102    #[serde(default = "default_cloud_threshold")]
103    pub cloud_threshold: f64,
104
105    /// Estimated cost per 1 000 input tokens for the local worker.
106    #[serde(default = "default_local_cost_per_1k_tokens")]
107    pub local_cost_per_1k_tokens: f64,
108
109    /// Estimated cost per 1 000 input tokens for the cloud worker.
110    #[serde(default = "default_cloud_cost_per_1k_tokens")]
111    pub cloud_cost_per_1k_tokens: f64,
112
113    /// Adaptive threshold tuning settings.
114    #[serde(default)]
115    pub adaptive: AdaptiveConfig,
116
117    /// Domain-specific terminology used by the complexity scorer.
118    ///
119    /// Any prompt containing two or more of these terms (case-insensitive)
120    /// receives the `+0.15` domain-term complexity bonus.
121    /// Defaults to a built-in list of Rust and financial terms.
122    #[serde(default = "default_domain_terms")]
123    pub domain_terms: Vec<String>,
124}
125
126impl Default for RoutingConfig {
127    fn default() -> Self {
128        Self {
129            local_threshold: default_local_threshold(),
130            cloud_threshold: default_cloud_threshold(),
131            local_cost_per_1k_tokens: default_local_cost_per_1k_tokens(),
132            cloud_cost_per_1k_tokens: default_cloud_cost_per_1k_tokens(),
133            adaptive: AdaptiveConfig::default(),
134            domain_terms: default_domain_terms(),
135        }
136    }
137}

139/// Adaptive threshold adjustment configuration.
140///
141/// When enabled the router raises or lowers the local/cloud boundary based
142/// on observed local-model failure rates in the fallback zone.
143///
144/// # Panics
145///
146/// This type never panics.
147#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
148pub struct AdaptiveConfig {
149    /// Whether adaptive threshold adjustment is enabled.
150    #[serde(default = "default_adaptive_enabled")]
151    pub enabled: bool,
152
153    /// How much to raise/lower the cloud threshold per adaptation step.
154    #[serde(default = "default_adaptive_step")]
155    pub step: f64,
156
157    /// Minimum completed fallback outcomes before the first adaptation.
158    #[serde(default = "default_adaptive_min_samples")]
159    pub min_samples: u64,
160
161    /// Local failure rate ceiling. If the observed failure rate in the
162    /// fallback zone exceeds this value the threshold is raised.
163    #[serde(default = "default_adaptive_failure_ceiling")]
164    pub failure_ceiling: f64,
165}
166
167impl Default for AdaptiveConfig {
168    fn default() -> Self {
169        Self {
170            enabled: default_adaptive_enabled(),
171            step: default_adaptive_step(),
172            min_samples: default_adaptive_min_samples(),
173            failure_ceiling: default_adaptive_failure_ceiling(),
174        }
175    }
176}

178/// Validate a [`RoutingConfig`], returning a list of human-readable errors.
179///
180/// # Arguments
181///
182/// * `config`  -  The routing configuration to validate.
183///
184/// # Returns
185///
186/// An empty `Vec` on success, or one error string per violated constraint.
187///
188/// # Panics
189///
190/// This function never panics.
191pub fn validate(config: &RoutingConfig) -> Vec<String> {
192    let mut errors = Vec::new();
193
194    if config.local_threshold < 0.0 || config.local_threshold > 1.0 {
195        errors.push(format!(
196            "local_threshold must be in [0.0, 1.0], got {}",
197            config.local_threshold
198        ));
199    }
200
201    if config.cloud_threshold < 0.0 || config.cloud_threshold > 1.0 {
202        errors.push(format!(
203            "cloud_threshold must be in [0.0, 1.0], got {}",
204            config.cloud_threshold
205        ));
206    }
207
208    if config.cloud_threshold < config.local_threshold {
209        errors.push(format!(
210            "cloud_threshold ({}) must be >= local_threshold ({})",
211            config.cloud_threshold, config.local_threshold
212        ));
213    }
214
215    if config.local_cost_per_1k_tokens < 0.0 {
216        errors.push(format!(
217            "local_cost_per_1k_tokens must be >= 0, got {}",
218            config.local_cost_per_1k_tokens
219        ));
220    }
221
222    if config.cloud_cost_per_1k_tokens < 0.0 {
223        errors.push(format!(
224            "cloud_cost_per_1k_tokens must be >= 0, got {}",
225            config.cloud_cost_per_1k_tokens
226        ));
227    }
228
229    if config.adaptive.step <= 0.0 || config.adaptive.step > 0.5 {
230        errors.push(format!(
231            "adaptive.step must be in (0.0, 0.5], got {}",
232            config.adaptive.step
233        ));
234    }
235
236    if config.adaptive.failure_ceiling <= 0.0 || config.adaptive.failure_ceiling > 1.0 {
237        errors.push(format!(
238            "adaptive.failure_ceiling must be in (0.0, 1.0], got {}",
239            config.adaptive.failure_ceiling
240        ));
241    }
242
243    errors
244}
