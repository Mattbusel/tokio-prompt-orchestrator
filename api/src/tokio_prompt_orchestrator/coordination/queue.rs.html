1//! # TaskQueue — file-locked concurrent task claiming
2//!
3//! ## Responsibility
4//! Provide atomic task claiming using filesystem lock files.
5//! Multiple agent processes can safely claim tasks without races.
6//!
7//! ## Guarantees
8//! - Atomic: task claiming uses `create_new(true)` for exclusivity
9//! - Stale-safe: locks older than configured timeout are reclaimable
10//! - Crash-safe: dead agent locks are detectable and stealable
11//! - Non-blocking: all file I/O runs via `spawn_blocking`
12//!
13//! ## NOT Responsible For
14//! - Process management (see: spawner.rs)
15//! - Task definitions (see: task.rs)
16//! - Health monitoring (see: monitor.rs)
17

25/// Lock file contents: agent ID and timestamp.
26#[derive(Debug, Clone)]
27struct LockInfo {
28    /// Agent that holds this lock.
29    agent_id: String,
30    /// When the lock was created (Unix timestamp seconds).
31    timestamp: u64,
32}
33
34impl std::fmt::Display for LockInfo {
35    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
36        write!(f, "{}\n{}", self.agent_id, self.timestamp)
37    }
38}
39
40impl LockInfo {
41    /// Parse from lock file contents.
42    ///
43    /// # Returns
44    ///
45    /// - `Some(LockInfo)` if parsing succeeds
46    /// - `None` if the format is invalid
47    fn from_str(s: &str) -> Option<Self> {
48        let mut lines = s.lines();
49        let agent_id = lines.next()?.to_string();
50        let timestamp: u64 = lines.next()?.parse().ok()?;
51        Some(Self {
52            agent_id,
53            timestamp,
54        })
55    }
56}

58/// File-locked task queue for concurrent agent coordination.
59///
60/// The queue reads tasks from a TOML file and uses per-task lock files
61/// in a dedicated directory for atomic claiming. Lock file presence
62/// determines task status at runtime:
63///
64/// - `{lock_dir}/{task_id}.claimed` — task is claimed by an agent
65/// - `{lock_dir}/{task_id}.completed` — task finished successfully
66/// - `{lock_dir}/{task_id}.failed` — task failed (contents: reason)
67///
68/// # Thread Safety
69///
70/// The queue is safe for concurrent access from multiple tasks within
71/// a single process. Cross-process safety comes from the filesystem locks.
72///
73/// # Panics
74///
75/// No methods on this type panic.
76///
77/// # Example
78///
79/// ```rust,no_run
80/// use tokio_prompt_orchestrator::coordination::queue::TaskQueue;
81/// use tokio_prompt_orchestrator::coordination::config::CoordinationConfig;
82/// use std::sync::Arc;
83///
84/// # async fn example() -> Result<(), Box<dyn std::error::Error>> {
85/// let config = CoordinationConfig::default();
86/// let queue = TaskQueue::new(Arc::new(config)).await?;
87/// if let Some(task) = queue.claim_next("agent-1").await? {
88///     println!("Claimed task: {}", task.id);
89/// }
90/// # Ok(())
91/// # }
92/// ```
93pub struct TaskQueue {
94    /// Shared configuration.
95    config: Arc<CoordinationConfig>,
96    /// In-memory task list (loaded from TOML, updated by claim/complete/fail).
97    tasks: Mutex<Vec<Task>>,
98    /// Lock directory path.
99    lock_dir: PathBuf,
100}
101
102impl TaskQueue {
103    /// Create a new TaskQueue, loading tasks from the configured TOML file.
104    ///
105    /// Creates the lock directory if it does not exist.
106    ///
107    /// # Arguments
108    ///
109    /// * `config` — Shared coordination configuration
110    ///
111    /// # Returns
112    ///
113    /// - `Ok(TaskQueue)` on success
114    /// - `Err(CoordinationError::TaskFileNotFound)` if the task file doesn't exist
115    /// - `Err(CoordinationError::TaskFileParse)` if the TOML is invalid
116    /// - `Err(CoordinationError::Io)` on filesystem errors
117    ///
118    /// # Panics
119    ///
120    /// This function never panics.
121    pub async fn new(config: Arc<CoordinationConfig>) -> Result<Self, CoordinationError> {
122        let task_file_path = config.task_file.clone();
123        let lock_dir = config.lock_dir.clone();
124
125        // Read task file
126        let content = tokio::task::spawn_blocking(move || std::fs::read_to_string(&task_file_path))
127            .await
128            .map_err(|e| CoordinationError::Io(std::io::Error::other(e)))?
129            .map_err(|e| {
130                if e.kind() == std::io::ErrorKind::NotFound {
131                    CoordinationError::TaskFileNotFound {
132                        path: config.task_file.clone(),
133                    }
134                } else {
135                    CoordinationError::Io(e)
136                }
137            })?;
138
139        let task_file = TaskFile::from_toml(&content)?;
140
141        // Create lock directory
142        let lock_dir_clone = lock_dir.clone();
143        tokio::task::spawn_blocking(move || std::fs::create_dir_all(&lock_dir_clone))
144            .await
145            .map_err(|e| CoordinationError::Io(std::io::Error::other(e)))??;
146
147        // Reconcile task statuses with existing lock files
148        let mut tasks = task_file.tasks;
149        for task in &mut tasks {
150            let status = Self::read_lock_status_sync(&lock_dir, &task.id);
151            match status {
152                Some((TaskStatus::Claimed, agent)) | Some((TaskStatus::Running, agent)) => {
153                    task.status = TaskStatus::Claimed;
154                    task.claimed_by = agent;
155                }
156                Some((TaskStatus::Completed, agent)) => {
157                    task.status = TaskStatus::Completed;
158                    task.claimed_by = agent;
159                }
160                Some((TaskStatus::Failed, agent)) => {
161                    task.status = TaskStatus::Failed;
162                    task.claimed_by = agent;
163                }
164                _ => {}
165            }
166        }
167
168        Ok(Self {
169            config,
170            tasks: Mutex::new(tasks),
171            lock_dir,
172        })
173    }

203    /// Atomically claim the next available task by priority.
204    ///
205    /// Writes a lock file `{lock_dir}/{task_id}.claimed` with the agent ID.
206    /// If a lock file already exists and is stale (older than `stale_lock_secs`),
207    /// the lock is stolen and the task is reclaimed.
208    ///
209    /// # Arguments
210    ///
211    /// * `agent_id` — Identifier of the claiming agent
212    ///
213    /// # Returns
214    ///
215    /// - `Ok(Some(task))` if a task was successfully claimed
216    /// - `Ok(None)` if no tasks are available
217    /// - `Err(CoordinationError)` on I/O failure
218    ///
219    /// # Panics
220    ///
221    /// This function never panics.
222    pub async fn claim_next(&self, agent_id: &str) -> Result<Option<Task>, CoordinationError> {
223        let mut tasks = self.tasks.lock().await;
224
225        // Consider both pending and claimed tasks (stale locks can be stolen)
226        let mut candidate_indices: Vec<usize> = tasks
227            .iter()
228            .enumerate()
229            .filter(|(_, t)| !t.status.is_terminal())
230            .map(|(i, _)| i)
231            .collect();
232
233        candidate_indices.sort_by_key(|&i| tasks[i].priority);
234
235        for idx in candidate_indices {
236            let task_id = tasks[idx].id.clone();
237            let lock_path = self.claimed_path(&task_id);
238
239            match self.try_create_lock(&lock_path, agent_id).await {
240                Ok(true) => {
241                    // Successfully claimed (no lock file existed)
242                    tasks[idx].status = TaskStatus::Claimed;
243                    tasks[idx].claimed_by = agent_id.to_string();
244                    return Ok(Some(tasks[idx].clone()));
245                }
246                Ok(false) => {
247                    // Lock exists — check if stale
248                    if self.is_lock_stale(&lock_path).await? {
249                        // Steal the lock from a dead/stale agent
250                        self.steal_lock(&lock_path, agent_id).await?;
251                        tasks[idx].status = TaskStatus::Claimed;
252                        tasks[idx].claimed_by = agent_id.to_string();
253                        return Ok(Some(tasks[idx].clone()));
254                    }
255                    // Not stale — skip this task, try next
256                    continue;
257                }
258                Err(e) => return Err(e),
259            }
260        }
261
262        Ok(None)
263    }

265    /// Mark a task as completed.
266    ///
267    /// Creates `{lock_dir}/{task_id}.completed` and removes the `.claimed` file.
268    ///
269    /// # Arguments
270    ///
271    /// * `task_id` — ID of the completed task
272    /// * `agent_id` — ID of the agent that completed the task
273    ///
274    /// # Returns
275    ///
276    /// - `Ok(())` on success
277    /// - `Err(CoordinationError::TaskNotFound)` if the task doesn't exist
278    ///
279    /// # Panics
280    ///
281    /// This function never panics.
282    pub async fn complete(&self, task_id: &str, agent_id: &str) -> Result<(), CoordinationError> {
283        let mut tasks = self.tasks.lock().await;
284
285        let task = tasks.iter_mut().find(|t| t.id == task_id).ok_or_else(|| {
286            CoordinationError::TaskNotFound {
287                id: task_id.to_string(),
288            }
289        })?;
290
291        task.status = TaskStatus::Completed;
292
293        // Write completion marker
294        let completed_path = self.completed_path(task_id);
295        let content = format!("{agent_id}\n{}", current_timestamp());
296        let claimed_path = self.claimed_path(task_id);
297
298        tokio::task::spawn_blocking(move || {
299            std::fs::write(&completed_path, content)?;
300            // Remove claimed lock (best-effort)
301            let _ = std::fs::remove_file(&claimed_path);
302            Ok::<(), std::io::Error>(())
303        })
304        .await
305        .map_err(|e| CoordinationError::Io(std::io::Error::other(e)))??;
306
307        Ok(())
308    }

310    /// Mark a task as failed.
311    ///
312    /// Creates `{lock_dir}/{task_id}.failed` with the failure reason.
313    ///
314    /// # Arguments
315    ///
316    /// * `task_id` — ID of the failed task
317    /// * `agent_id` — ID of the agent that reported the failure
318    /// * `reason` — Human-readable failure reason
319    ///
320    /// # Returns
321    ///
322    /// - `Ok(())` on success
323    /// - `Err(CoordinationError::TaskNotFound)` if the task doesn't exist
324    ///
325    /// # Panics
326    ///
327    /// This function never panics.
328    pub async fn fail(
329        &self,
330        task_id: &str,
331        agent_id: &str,
332        reason: &str,
333    ) -> Result<(), CoordinationError> {
334        let mut tasks = self.tasks.lock().await;
335
336        let task = tasks.iter_mut().find(|t| t.id == task_id).ok_or_else(|| {
337            CoordinationError::TaskNotFound {
338                id: task_id.to_string(),
339            }
340        })?;
341
342        task.status = TaskStatus::Failed;
343        task.failure_reason = reason.to_string();
344
345        let failed_path = self.failed_path(task_id);
346        let content = format!("{agent_id}\n{}\n{reason}", current_timestamp());
347        let claimed_path = self.claimed_path(task_id);
348
349        tokio::task::spawn_blocking(move || {
350            std::fs::write(&failed_path, content)?;
351            let _ = std::fs::remove_file(&claimed_path);
352            Ok::<(), std::io::Error>(())
353        })
354        .await
355        .map_err(|e| CoordinationError::Io(std::io::Error::other(e)))??;
356
357        Ok(())
358    }

360    /// Release a claimed task back to pending status.
361    ///
362    /// Removes the `.claimed` lock file so other agents can claim it.
363    ///
364    /// # Arguments
365    ///
366    /// * `task_id` — ID of the task to release
367    ///
368    /// # Returns
369    ///
370    /// - `Ok(())` on success
371    /// - `Err(CoordinationError::TaskNotFound)` if the task doesn't exist
372    ///
373    /// # Panics
374    ///
375    /// This function never panics.
376    pub async fn release(&self, task_id: &str) -> Result<(), CoordinationError> {
377        let mut tasks = self.tasks.lock().await;
378
379        let task = tasks.iter_mut().find(|t| t.id == task_id).ok_or_else(|| {
380            CoordinationError::TaskNotFound {
381                id: task_id.to_string(),
382            }
383        })?;
384
385        task.status = TaskStatus::Pending;
386        task.claimed_by.clear();
387
388        let claimed_path = self.claimed_path(task_id);
389        tokio::task::spawn_blocking(move || {
390            let _ = std::fs::remove_file(&claimed_path);
391        })
392        .await
393        .map_err(|e| CoordinationError::Io(std::io::Error::other(e)))?;
394
395        Ok(())
396    }

408    /// Get summary counts: (pending, claimed, completed, failed).
409    ///
410    /// # Panics
411    ///
412    /// This function never panics.
413    pub async fn summary(&self) -> (usize, usize, usize, usize) {
414        let tasks = self.tasks.lock().await;
415        let pending = tasks
416            .iter()
417            .filter(|t| t.status == TaskStatus::Pending)
418            .count();
419        let claimed = tasks
420            .iter()
421            .filter(|t| t.status == TaskStatus::Claimed || t.status == TaskStatus::Running)
422            .count();
423        let completed = tasks
424            .iter()
425            .filter(|t| t.status == TaskStatus::Completed)
426            .count();
427        let failed = tasks
428            .iter()
429            .filter(|t| t.status == TaskStatus::Failed)
430            .count();
431        (pending, claimed, completed, failed)
432    }

461    /// Try to atomically create a lock file. Returns Ok(true) if created,
462    /// Ok(false) if the file already exists.
463    async fn try_create_lock(
464        &self,
465        path: &Path,
466        agent_id: &str,
467    ) -> Result<bool, CoordinationError> {
468        let path = path.to_path_buf();
469        let content = LockInfo {
470            agent_id: agent_id.to_string(),
471            timestamp: current_timestamp(),
472        }
473        .to_string();
474
475        let result = tokio::task::spawn_blocking(move || {
476            use std::fs::OpenOptions;
477            use std::io::Write;
478
479            match OpenOptions::new().write(true).create_new(true).open(&path) {
480                Ok(mut file) => {
481                    file.write_all(content.as_bytes())?;
482                    Ok(true)
483                }
484                Err(e) if e.kind() == std::io::ErrorKind::AlreadyExists => Ok(false),
485                Err(e) => Err(e),
486            }
487        })
488        .await
489        .map_err(|e| CoordinationError::Io(std::io::Error::other(e)))??;
490
491        Ok(result)
492    }

494    /// Check if a lock file is stale (older than stale_lock_secs).
495    async fn is_lock_stale(&self, path: &Path) -> Result<bool, CoordinationError> {
496        let path = path.to_path_buf();
497        let stale_secs = self.config.stale_lock_secs;
498
499        let result = tokio::task::spawn_blocking(move || -> Result<bool, std::io::Error> {
500            let content = match std::fs::read_to_string(&path) {
501                Ok(c) => c,
502                Err(e) if e.kind() == std::io::ErrorKind::NotFound => return Ok(false),
503                Err(e) => return Err(e),
504            };
505
506            if let Some(info) = LockInfo::from_str(&content) {
507                let now = current_timestamp();
508                // Use >= so that stale_lock_secs=0 means "immediately stale"
509                // (a 0-second-old lock is stale when the threshold is 0).
510                Ok(now.saturating_sub(info.timestamp) >= stale_secs)
511            } else {
512                // Corrupt lock file — treat as stale
513                Ok(true)
514            }
515        })
516        .await
517        .map_err(|e| CoordinationError::Io(std::io::Error::other(e)))??;
518
519        Ok(result)
520    }

522    /// Steal a stale lock by removing and recreating it.
523    async fn steal_lock(&self, path: &Path, agent_id: &str) -> Result<(), CoordinationError> {
524        let path = path.to_path_buf();
525        let content = LockInfo {
526            agent_id: agent_id.to_string(),
527            timestamp: current_timestamp(),
528        }
529        .to_string();
530
531        tokio::task::spawn_blocking(move || {
532            let _ = std::fs::remove_file(&path);
533            std::fs::write(&path, content)
534        })
535        .await
536        .map_err(|e| CoordinationError::Io(std::io::Error::other(e)))??;
537
538        Ok(())
539    }

541    /// Read lock status from the filesystem (synchronous, for init).
542    fn read_lock_status_sync(lock_dir: &Path, task_id: &str) -> Option<(TaskStatus, String)> {
543        // Check completed first
544        let completed_path = lock_dir.join(format!("{task_id}.completed"));
545        if let Ok(content) = std::fs::read_to_string(&completed_path) {
546            let agent = content.lines().next().unwrap_or_default().to_string();
547            return Some((TaskStatus::Completed, agent));
548        }
549
550        // Check failed
551        let failed_path = lock_dir.join(format!("{task_id}.failed"));
552        if let Ok(content) = std::fs::read_to_string(&failed_path) {
553            let agent = content.lines().next().unwrap_or_default().to_string();
554            return Some((TaskStatus::Failed, agent));
555        }
556
557        // Check claimed
558        let claimed_path = lock_dir.join(format!("{task_id}.claimed"));
559        if let Ok(content) = std::fs::read_to_string(&claimed_path) {
560            if let Some(info) = LockInfo::from_str(&content) {
561                return Some((TaskStatus::Claimed, info.agent_id));
562            }
563        }
564
565        None
566    }

569/// Get the current Unix timestamp in seconds.
570fn current_timestamp() -> u64 {
571    std::time::SystemTime::now()
572        .duration_since(std::time::UNIX_EPOCH)
573        .unwrap_or_default()
574        .as_secs()
575}
