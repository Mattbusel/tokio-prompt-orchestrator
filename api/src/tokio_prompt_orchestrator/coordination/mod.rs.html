1//! # Coordination — programmatic agent fleet management
2//!
3//! ## Responsibility
4//! Provide a complete agent coordination layer: define tasks in TOML,
5//! spawn agent processes, let them claim work atomically via filesystem
6//! locks, and monitor fleet health — all without manual terminal pane
7//! management.
8//!
9//! ## Architecture
10//!
11//! ```text
12//! tasks.toml ──► TaskQueue (file-locked claiming)
13//!                    │
14//!                    ├──► AgentWorker 0 ──► claude process
15//!                    ├──► AgentWorker 1 ──► claude process
16//!                    ├──► AgentWorker N ──► claude process
17//!                    │
18//!               AgentMonitor (health checks, status)
19//! ```
20//!
21//! ## Modules
22//!
23//! - [`task`] — Task struct, TaskStatus, priority ordering
24//! - [`config`] — CoordinationConfig with defaults and validation
25//! - [`queue`] — TaskQueue with file-locked concurrent claiming
26//! - [`worker`] — AgentWorker: spawn claude process, capture output
27//! - [`spawner`] — AgentSpawner: spawn N agents, manage fleet lifecycle
28//! - [`monitor`] — AgentMonitor: health checks, fleet status snapshots
29//!
30//! ## Guarantees
31//!
32//! - **Atomic claiming**: no task is ever executed twice (filesystem locks)
33//! - **Crash recovery**: stale locks from dead agents are reclaimable
34//! - **Priority ordering**: highest-priority tasks are claimed first
35//! - **Observable**: fleet status available at any time via monitor snapshots
36//! - **Zero panics**: all operations return `Result`, never panic
37//!
38//! ## NOT Responsible For
39//!
40//! - LLM inference (see: `worker` module at crate root)
41//! - Pipeline stages (see: `stages` module)
42//! - Web API (see: `web_api` module)
43//!
44//! ## Example
45//!
46//! ```rust,no_run
47//! use tokio_prompt_orchestrator::coordination::{
48//!     config::CoordinationConfig,
49//!     queue::TaskQueue,
50//!     spawner::{AgentSpawner, await_fleet},
51//! };
52//! use std::sync::Arc;
53//!
54//! # async fn example() -> Result<(), Box<dyn std::error::Error>> {
55//! let config = Arc::new(CoordinationConfig::default());
56//! let queue = Arc::new(TaskQueue::new(config.clone()).await?);
57//! let spawner = AgentSpawner::new(config);
58//! let handles = spawner.spawn_fleet(queue).await?;
59//! let stats = await_fleet(handles).await;
60//! for s in &stats {
61//!     println!("{}: {} completed, {} failed", s.agent_id, s.tasks_completed, s.tasks_failed);
62//! }
63//! # Ok(())
64//! # }
65//! ```

77/// Errors specific to the agent coordination layer.
78///
79/// All variants implement `std::error::Error` via [`thiserror`].
80/// Every variant has at least one test that triggers it.
81///
82/// # Panics
83///
84/// No methods on this type panic.
85#[derive(Error, Debug)]
86pub enum CoordinationError {
87    /// Task file not found at the specified path.
88    #[error("task file not found: {}", path.display())]
89    TaskFileNotFound {
90        /// Path that was looked up.
91        path: PathBuf,
92    },
93
94    /// Task file could not be parsed as valid TOML.
95    #[error("task file parse error: {0}")]
96    TaskFileParse(String),
97
98    /// Referenced task ID does not exist in the queue.
99    #[error("task not found: {id}")]
100    TaskNotFound {
101        /// The task ID that was not found.
102        id: String,
103    },
104
105    /// Task was already claimed by another agent.
106    #[error("task already claimed: {id} by {agent}")]
107    TaskAlreadyClaimed {
108        /// The task ID.
109        id: String,
110        /// The agent that holds the claim.
111        agent: String,
112    },
113
114    /// Agent process could not be spawned.
115    #[error("agent spawn failed: {0}")]
116    SpawnError(String),
117
118    /// Agent exceeded its configured timeout.
119    #[error("agent timeout: agent {agent_id} exceeded {timeout_secs}s")]
120    AgentTimeout {
121        /// The agent that timed out.
122        agent_id: String,
123        /// The timeout that was exceeded.
124        timeout_secs: u64,
125    },
126
127    /// Filesystem I/O error.
128    #[error("io error: {0}")]
129    Io(#[from] std::io::Error),
130
131    /// No tasks are available for claiming.
132    #[error("no tasks available")]
133    NoTasksAvailable,
134
135    /// Configuration validation failed.
136    #[error("invalid config: {0}")]
137    InvalidConfig(String),
138}

144    #[test]
145    fn test_coordination_error_display_task_file_not_found() {
146        let err = CoordinationError::TaskFileNotFound {
147            path: PathBuf::from("tasks.toml"),
148        };
149        assert!(err.to_string().contains("tasks.toml"));
150    }

153    fn test_coordination_error_display_task_file_parse() {
154        let err = CoordinationError::TaskFileParse("unexpected token".to_string());
155        assert!(err.to_string().contains("unexpected token"));
156    }

164    }

166    #[test]
167    fn test_coordination_error_display_task_already_claimed() {
168        let err = CoordinationError::TaskAlreadyClaimed {
169            id: "task-1".to_string(),

172        let msg = err.to_string();

174        assert!(msg.contains("agent-0"));
